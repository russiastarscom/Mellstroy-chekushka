   227→   227→  function mkParticle(x, y, kind) {

   233→   233→      g: kind === 'dust' ? 120 : 300,
   234→   234→      life: 0.5 + Math.random() * 0.4, maxLife: 0.9,

   236→   236→      color: colors[kind] || '#fff',

   242→   242→  // ---------- обновление ----------
   243→   243→  function update(dt) {
   244→   244→    tGlobal += dt;
   245→   245→    if (shake > 0) shake -= dt;
   246→   246→
   247→   247→    if (state !== 'playing' || frozen) { input.jumpPressed = false; return; }
   248→   248→

   250→   250→

   252→   252→    if (player.onGround) {
   253→   253→      const fr = Math.floor((player.y + player.h + 2) / TILE);
   254→   254→      const c0 = Math.floor(player.x / TILE), c1 = Math.floor((player.x + player.w - 1) / TILE);
   255→   255→      for (let c = c0; c <= c1; c++) {
   256→   256→        if (tileAt(level, c, fr) === 'C' && !level.crumble.has(c + ',' + fr)) {
   257→   257→          level.crumble.set(c + ',' + fr, { t: 0, broken: false });
   258→   258→          Audio8.sfx.crumble();
   259→   259→        }
   260→   260→      }
   261→   261→    }
   262→   262→    level.crumble.forEach((st, k) => {
   263→   263→      st.t += dt;
   264→   264→      if (!st.broken && st.t > 0.45) {
   265→   265→        st.broken = true; st.t = 0;
   266→   266→        Audio8.sfx.crumble();
   267→   267→        const [cc, rr] = k.split(',').map(Number);
   268→   268→        spawnDust(cc * TILE + 20, rr * TILE + 20, 6);
   269→   269→      } else if (st.broken && st.t > 2.6) {
   270→   270→        level.crumble.delete(k);
   271→   271→      }
   272→   272→    });
   273→   273→
   274→   274→    // ---------- плюшки: бросок ----------
   275→   275→    if (player.shootCd > 0) player.shootCd -= dt;
   276→   276→    if (input.shootPressed) {
   277→   277→      input.shootPressed = false;
   278→   278→      if (player.ammo > 0 && player.shootCd <= 0) {
   279→   279→        player.ammo--;
   280→   280→        player.shootCd = 0.45;
   281→   281→        const dir = player.dir || 1;
   282→   282→        const hx = dir > 0 ? player.x + player.w - 4 : player.x - 26;
   283→   283→        hatShots.push(new HatShot(hx, player.y + 2, dir));
   284→   284→        Audio8.sfx.throw();
   285→   285→      } else if (player.ammo <= 0) {
   286→   286→        Audio8.sfx.locked(); // пусто — щёлкаем
   287→   287→      }
   288→   288→    }
   289→   289→
   290→   290→    enemies.forEach((e) => e.update(dt, level, api));
   291→   291→    tomahawks.forEach((t) => t.update(dt, level, api));
   292→   292→    tomahawks = tomahawks.filter((t) => !t.dead);
   293→   293→
   294→   294→    // столкновения игрок-враги
   295→   295→    enemies.forEach((e) => {
   296→   296→      if (e.dead) return;
   297→   297→      if (!overlaps(player, e)) return;
   298→   298→      const stomping = player.vy > 90 && (player.y + player.h) - e.y < 20;
   299→   299→      if (stomping) {
   300→   300→        if (e instanceof Boss) {
   301→   301→          if (e.invuln > 0) { player.vy = -480; }
   302→   302→          else { e.stomp(api); player.vy = -540; }
   303→   303→        } else {
   304→   304→          e.stomp(api);
   305→   305→          player.vy = -490;
   306→   306→        }
   307→   307→      } else {
   308→   308→        player.hurt(Math.sign(player.x - e.x) || 1, api);
   309→   309→      }
   310→   310→    });
   311→   311→    enemies = enemies.filter((e) => !e.dead);
   312→   312→
   313→   313→    // предметы
   314→   314→    bottles.forEach((b) => {
   315→   315→      b.update(dt);
   316→   316→      if (!b.taken && overlaps(player, b.rect())) {
   317→   317→        b.taken = true;
   318→   318→        bottlesGot++;
   319→   319→        Audio8.sfx.coin();
   320→   320→        for (let i = 0; i < 4; i++) particles.push(mkParticle(b.c * TILE + 20, b.r * TILE + 20, 'sparkle'));
   321→   321→      }
   322→   322→    });
   323→   323→    bottles = bottles.filter((b) => !b.taken);
   324→   324→    hearts.forEach((h) => {
   325→   325→      h.update(dt);
   326→   326→      if (!h.taken && overlaps(player, h.rect())) {
   327→   327→        h.taken = true;
   328→   328→        if (player.hp < CONFIG.PLAYER_HP) { player.hp++; Audio8.sfx.heart(); }
   329→   329→        else { bottlesGot++; Audio8.sfx.coin(); }
   330→   330→      }
   331→   331→    });
   332→   332→    hearts = hearts.filter((h) => !h.taken);
   333→   333→
   334→   334→    // плюшки — подбор боеприпаса
   335→   335→    plushes.forEach((p) => {
   336→   336→      p.update(dt);
   337→   337→      if (!p.taken && overlaps(player, p.rect())) {
   338→   338→        p.taken = true;
   339→   339→        player.ammo = Math.min(9, player.ammo + 3);
   340→   340→        Audio8.sfx.heart();
   341→   341→        for (let i = 0; i < 6; i++) particles.push(mkParticle(p.c * TILE + 20, p.r * TILE + 20, 'sparkle'));
   342→   342→      }
   343→   343→    });
   344→   344→    plushes = plushes.filter((p) => !p.taken);
   345→   345→
   346→   346→    // кастомные объекты (админ-панель): скрипт + взаимодействия с игроком
   347→   347→    customEnts.forEach((o) => o.update(dt, level, scriptApi));
   348→   348→    customEnts.forEach((o) => {
   349→   349→      if (o.dead || o.taken) return;
   350→   350→      if (!overlaps(player, o)) return;
   351→   351→      const stomping = player.vy > 90 && (player.y + player.h) - o.y < 24;
   352→   352→      if (o.collect > 0) {
   353→   353→        o.taken = true;
   354→   354→        bottlesGot += o.collect;
   355→   355→        Audio8.sfx.coin();
   356→   356→        scriptApi.particles(o.x + o.w / 2, o.y + o.h / 2, 'sparkle', 5);
   357→   357→      } else if (stomping && o.stompable) {
   358→   358→        o.hurt(api);
   359→   359→        player.vy = -490;
   360→   360→      } else if (o.dangerous) {
   361→   361→        player.hurt(Math.sign(player.x - o.x) || 1, api);
   362→   362→      }
   363→   363→    });
   364→   364→    customEnts = customEnts.filter((o) => !o.dead && !o.taken);
   365→   365→
   366→   366→    // летящие плюшки — столкновения
   367→   367→    hatShots.forEach((s) => s.update(dt, level, api));
   368→   368→    hatShots.forEach((s) => {

   381→   381→      if (!s.dead) {
   382→   382→        for (const o of customEnts) {
   383→   383→          if (o.dead || o.taken || !o.stompable) continue;

   390→   390→    // завод — выход (триггер = всё здание)

   393→   393→

   395→   395→    particles.forEach((p) => { p.vy += p.g * dt; p.x += p.vx * dt; p.y += p.vy * dt; p.life -= dt; });

   402→   402→    const ty = Math.max(0, Math.min(player.y + player.h / 2 - viewH * 0.55, VIEW_H - viewH));
   403→   403→    camY += (ty - camY) * Math.min(1, dt * 7);
   404→   404→
   405→   405→    // HUD
   406→   406→    if (hudCache.hp !== player.hp || hudCache.bottles !== bottlesGot || hudCache.hats !== player.ammo) {
   407→   407→      hudCache = { hp: player.hp, bottles: bottlesGot, hats: player.ammo };
   408→   408→      UI.setHud({ hp: player.hp, bottles: bottlesGot, hats: player.ammo });
   409→   409→    }
   410→   410→
   411→   411→    input.jumpPressed = false;
   412→   412→  }

   580→   580→  function render() {
   581→   581→    ctx.clearRect(0, 0, VIEW_W, VIEW_H);
   582→   582→    if (!level) return;

   590→   590→    ctx.translate(sx, sy);
   591→   591→    // приближение: мир рисуется крупнее, камера следит за Андреем

   593→   593→    ctx.translate(0, -camY);

   599→   599→    hearts.forEach((h) => h.draw(ctx, cam));
   600→   600→    plushes.forEach((p) => p.draw(ctx, cam));
   601→   601→    factory.draw(ctx, cam);
   602→   602→    enemies.forEach((e) => e.draw(ctx, cam, tGlobal));
   603→   603→    tomahawks.forEach((t) => t.draw(ctx, cam));
   604→   604→    hatShots.forEach((s) => s.draw(ctx, cam));
   605→   605→    customEnts.forEach((o) => o.draw(ctx, cam));
   606→   606→    if (player) player.draw(ctx, cam);
   607→   607→
   608→   608→    // частицы
   609→   609→    particles.forEach((p) => {
   610→   610→      ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
   611→   611→      ctx.fillStyle = p.color;
   612→   612→      ctx.fillRect(p.x - cam - p.size / 2, p.y - p.size / 2, p.size, p.size);
   613→   613→      ctx.globalAlpha = 1;
   614→   614→    });
   615→   615→
   616→   616→    ctx.restore();
   617→   617→  }

   690→   690→      }
   691→   691→      if (e.code === 'KeyP' || e.code === 'Escape') togglePause();
   692→   692→      if (e.code === 'KeyF' && !e.repeat) enterFullscreen();
   693→   693→    });
   694→   694→
   695→   695→    // тап/клик по экрану игры — ТОЛЬКО открыть полный экран (без выключения)

   698→   698→      enterFullscreen();

   775→   775→  function applyCms(j) {
   776→   776→    // кастомные объекты: текстуры + JS-скрипты (загружаем первыми — нужны картам)
   777→   777→    if (Array.isArray(j.objects) && j.objects.length) {
   778→   778→      CustomObjects.set(j.objects);
   779→   779→      j.objects.forEach((o) => {
   780→   780→        if (!o || !o.id) return;
   781→   781→        if (o.character) {
   782→   782→          CONFIG.NAMES['@' + o.id] = o.name || o.id;
   783→   783→          UI.addSpeaker('@' + o.id, 'obj-' + o.id);
   784→   784→        }
   785→   785→      });
   786→   786→    }
   787→   787→    if (Array.isArray(j.maps) && j.maps.length) {
   788→   788→      const intro = LEVELS[0];
   789→   789→      const defs = j.maps.map((m) => m && m.def).filter((d) => d && (d.type === 'map' || d.type === 'indev'));
   790→   790→      if (defs.length) {
   791→   791→        // старые публикации (без маркера v) всегда заканчивались экраном «В разработке» — сохраняем поведение
   792→   792→        if (!(j.v >= 3) && !defs.some((d) => d.type === 'indev')) {
   793→   793→          const iv = LEVELS.find((x) => x.type === 'indev');
   794→   794→          if (iv) defs.push(JSON.parse(JSON.stringify(iv)));
   795→   795→        }
   796→   796→        LEVELS.splice(0, LEVELS.length, intro, ...defs);
   797→   797→      }
   798→   798→    } else if (j.levels) {
   799→   799→      applyLegacyLevels(j.levels);
   800→   800→    }
   801→   801→    if (j.intro && typeof j.intro === 'object') {
   802→   802→      if (Array.isArray(j.intro.cutscene) && j.intro.cutscene.length && LEVELS[0]) LEVELS[0].cutscene = j.intro.cutscene;
   803→   803→      if (Array.isArray(j.intro.outroAfterBoss) && j.intro.outroAfterBoss.length) DIALOGUES.outroAfterBoss = j.intro.outroAfterBoss;
   804→   804→    }
   805→   805→    if (j.textures && typeof j.textures === 'object') {
   806→   806→      const map = {};
   807→   807→      Object.entries(j.textures).forEach(([k, v]) => { if (v && v.url) map[k] = v.url; });
   808→   808→      Sprites.setCustom(map);
   809→   809→      const bi = document.getElementById('hud-bottle-icon');
   810→   810→      if (bi && map.checkushka) { bi.style.visibility = ''; bi.src = map.checkushka; }
   811→   811→      const pi = document.getElementById('hud-plush-icon');
   812→   812→      if (pi && map.plush) { pi.style.visibility = ''; pi.src = map.plush; }
   813→   813→    }
   814→   814→    if (j.music && Array.isArray(j.music.tracks) && j.music.active) {
   815→   815→      const tr = j.music.tracks.find((t) => t.id === j.music.active);
   816→   816→      if (tr && tr.url) Audio8.setCustomMusic(tr.url);

   819→   819→      CONFIG.SOCIALS = j.socials.map((s) => ({ name: s.name, class: s.class || 'tg', url: s.url || '#' }));
