     4→     4→const TILE_SOLID = new Set(['#', 'B']);
     5→     5→
     6→     6→function isSolid(level, c, r) {
     7→     7→  if (c < 0 || c >= level.w) return true;   // стены по краям уровня
     8→     8→  if (r < 0 || r >= ROWS) return false;     // сверху/внизу — открыто
     9→     9→  return TILE_SOLID.has(level.grid[r][c]);
    10→    10→}
    11→    11→function isPlat(level, c, r) {
    12→    12→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return false;
    13→    13→  const t = level.grid[r][c];
    14→    14→  if (t === '=') return true;
    15→    15→  // сыпучая платформа: твёрдая, пока не обрушилась
    16→    16→  if (t === 'C') {
    17→    17→    const st = level.crumble && level.crumble.get(c + ',' + r);
    18→    18→    return !(st && st.broken);
    19→    19→  }
    20→    20→  return false;
    21→    21→}
    22→    22→function isSpike(level, c, r) {
    23→    23→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return false;
    24→    24→  return level.grid[r][c] === '^';
    25→    25→}
    26→    26→function tileAt(level, c, r) {
    27→    27→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return '.';
    28→    28→  return level.grid[r][c];
    29→    29→}
    30→    30→function isLiquid(level, c, r) { return tileAt(level, c, r) === 'L'; }

    32→    32→// Движение с разрешением коллизий по тайлам
    33→    33→function moveEntity(ent, level, dt, ignorePlats = false) {
    34→    34→  // --- X ---
    35→    35→  ent.x += ent.vx * dt;
    36→    36→  ent.hitWall = false;
    37→    37→  {
    38→    38→    const r0 = Math.floor(ent.y / TILE), r1 = Math.floor((ent.y + ent.h - 1) / TILE);
    39→    39→    const c1 = Math.floor((ent.x + ent.w - 1) / TILE), c0 = Math.floor(ent.x / TILE);
    40→    40→    if (ent.vx > 0) {
    41→    41→      for (let r = r0; r <= r1; r++) if (isSolid(level, c1, r)) { ent.x = c1 * TILE - ent.w; ent.vx = 0; ent.hitWall = true; break; }
    42→    42→    } else if (ent.vx < 0) {
    43→    43→      for (let r = r0; r <= r1; r++) if (isSolid(level, c0, r)) { ent.x = (c0 + 1) * TILE; ent.vx = 0; ent.hitWall = true; break; }
    44→    44→    }
    45→    45→  }
    46→    46→  // --- Y ---
    47→    47→  const prevBottom = ent.y + ent.h;
    48→    48→  ent.y += ent.vy * dt;
    49→    49→  ent.onGround = false;
    50→    50→  {
    51→    51→    const c0 = Math.floor(ent.x / TILE), c1 = Math.floor((ent.x + ent.w - 1) / TILE);
    52→    52→    const r0 = Math.floor(ent.y / TILE), r1 = Math.floor((ent.y + ent.h - 1) / TILE);
    53→    53→    if (ent.vy > 0) {
    54→    54→      for (let c = c0; c <= c1; c++) {
    55→    55→        if (isSolid(level, c, r1)) { ent.y = r1 * TILE - ent.h; ent.vy = 0; ent.onGround = true; break; }
    56→    56→        if (!ignorePlats && isPlat(level, c, r1) && prevBottom <= r1 * TILE + 8) {
    57→    57→          ent.y = r1 * TILE - ent.h; ent.vy = 0; ent.onGround = true; break;
    58→    58→        }
    59→    59→      }
    60→    60→    } else if (ent.vy < 0) {
    61→    61→      for (let c = c0; c <= c1; c++) {
    62→    62→        if (isSolid(level, c, r0)) { ent.y = (r0 + 1) * TILE; ent.vy = 0; break; }
    63→    63→      }
    64→    64→    }
    65→    65→  }
    66→    66→}

    68→    68→function overlaps(a, b) {
    69→    69→  return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
    70→    70→}

    81→    81→    this.dir = 1;
    82→    82→    this.onGround = false;
    83→    83→    this.hp = CONFIG.PLAYER_HP;
    84→    84→    this.invuln = 0;
    85→    85→    this.jumpBuffer = 0;
    86→    86→    this.coyote = 0;
    87→    87→    this.dropTimer = 0;
    88→    88→    this.lastSafe = { x: this.x, y: this.y };
    89→    89→    this.safeTimer = 0;
    90→    90→    this.walkT = 0;
    91→    91→    this.squash = 0;

    93→    93→
    94→    94→  update(dt, input, level, game) {
    95→    95→    const SPEED = CONFIG.MOVE_SPEED, ACC = 2200, FRICTION = 2600;
    96→    96→
    97→    97→    // горизонталь
    98→    98→    if (input.left) { this.vx -= ACC * dt; this.dir = -1; }
    99→    99→    if (input.right) { this.vx += ACC * dt; this.dir = 1; }
   100→   100→    if (!input.left && !input.right) {
   101→   101→      const s = Math.sign(this.vx);
   102→   102→      this.vx -= s * FRICTION * dt;
   103→   103→      if (Math.sign(this.vx) !== s) this.vx = 0;
   104→   104→    }
   105→   105→    this.vx = Math.max(-SPEED, Math.min(SPEED, this.vx));
   106→   106→
   107→   107→    // прыжок: буфер + койот
   108→   108→    this.jumpBuffer = input.jumpPressed ? 0.12 : Math.max(0, this.jumpBuffer - dt);
   109→   109→    this.coyote = this.onGround ? 0.1 : Math.max(0, this.coyote - dt);
   110→   110→    this.dropTimer = Math.max(0, this.dropTimer - dt);
   111→   111→
   112→   112→    if (this.jumpBuffer > 0 && input.down && this.onGround) {
   113→   113→      // спрыгнуть с платформы
   114→   114→      this.dropTimer = 0.22;
   115→   115→      this.jumpBuffer = 0;
   116→   116→      this.vy = 60;
   117→   117→    } else if (this.jumpBuffer > 0 && (this.coyote > 0)) {
   118→   118→      this.vy = CONFIG.JUMP_VEL;
   119→   119→      this.jumpBuffer = 0;
   120→   120→      this.coyote = 0;
   121→   121→      this.squash = -0.25;
   122→   122→      Audio8.sfx.jump();
   123→   123→      game.spawnDust(this.x + this.w / 2, this.y + this.h, 4);
   124→   124→    }
   125→   125→    // короткий прыжок при отпускании (батут не режем — у него свой импульс-таймер)
   126→   126→    this.springT = Math.max(0, (this.springT || 0) - dt);
   127→   127→    if (!input.jumpHeld && this.springT <= 0 && this.vy < -260) this.vy = -260;
   128→   128→
   129→   129→    this.vy = Math.min(this.vy + CONFIG.GRAVITY * dt, 1100);
   130→   130→    moveEntity(this, level, dt, this.dropTimer > 0);
   131→   131→
   132→   132→    // приземление — пыль + сквош
   133→   133→    if (this.onGround && this.prevVy > 380) {
   134→   134→      this.squash = 0.3;
   135→   135→      game.spawnDust(this.x + this.w / 2, this.y + this.h, 3);
   136→   136→    }
   137→   137→    this.prevVy = this.vy;
   138→   138→    this.squash *= Math.pow(0.001, dt);
   139→   139→
   140→   140→    // безопасная точка (для респауна после ямы)
   141→   141→    this.safeTimer -= dt;
   142→   142→    if (this.onGround && this.safeTimer <= 0 && this.vy === 0) {
   143→   143→      const fc = Math.floor((this.x + this.w / 2) / TILE);
   144→   144→      const fr = Math.floor((this.y + this.h + 6) / TILE);
   145→   145→      if (isSolid(level, fc, fr)) { this.lastSafe = { x: this.x, y: this.y }; this.safeTimer = 0.4; }
   146→   146→    }
   147→   147→
   148→   148→    // шипы
   149→   149→    if (this.invuln <= 0) {
   150→   150→      const c0 = Math.floor((this.x + 6) / TILE), c1 = Math.floor((this.x + this.w - 6) / TILE);
   151→   151→      const r = Math.floor((this.y + this.h - 4) / TILE);
   152→   152→      for (let c = c0; c <= c1; c++) {
   153→   153→        if (isSpike(level, c, r)) { this.hurt(c * TILE + 20 < this.x + this.w / 2 ? 1 : -1, game, true); break; }
   154→   154→      }
   155→   155→    }
   156→   156→
   157→   157→    // жидкость (полынья/лава/кислота): касание = урон + респавн, как падение в яму
   158→   158→    if (this.invuln <= 0) {
   159→   159→      const c0 = Math.floor((this.x + 4) / TILE), c1 = Math.floor((this.x + this.w - 4) / TILE);
   160→   160→      const r0 = Math.floor((this.y + 8) / TILE), r1 = Math.floor((this.y + this.h - 1) / TILE);
   161→   161→      outer: for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++) {
   162→   162→        if (isLiquid(level, c, r)) { game.playerFell(); return; }
   163→   163→      }
   164→   164→    }
   165→   165→
   166→   166→    // пилы: круглая пила в тайле — задевание с любой стороны
   167→   167→    if (this.invuln <= 0) {
   168→   168→      const c0 = Math.floor((this.x - 10) / TILE), c1 = Math.floor((this.x + this.w + 10) / TILE);
   169→   169→      const r0 = Math.floor((this.y - 10) / TILE), r1 = Math.floor((this.y + this.h + 10) / TILE);
   170→   170→      for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++) {
   171→   171→        if (tileAt(level, c, r) !== 'S') continue;
   172→   172→        const sx = c * TILE + 20, sy = r * TILE + 26, R = 24;
   173→   173→        const nx = Math.max(this.x, Math.min(sx, this.x + this.w));
   174→   174→        const ny = Math.max(this.y, Math.min(sy, this.y + this.h));
   175→   175→        if ((nx - sx) * (nx - sx) + (ny - sy) * (ny - sy) < R * R) {
   176→   176→          this.hurt(sx < this.x + this.w / 2 ? 1 : -1, game, true); break;
   177→   177→        }
   178→   178→      }
   179→   179→    }
   180→   180→
   181→   181→    // батут: приземлился/наступил — подброс (тайл батута в клетке ног или под ногами)
   182→   182→    if (this.onGround && this.vy === 0) {
   183→   183→      const sc0 = Math.floor((this.x + 4) / TILE), sc1 = Math.floor((this.x + this.w - 4) / TILE);
   184→   184→      const srA = Math.floor((this.y + this.h - 2) / TILE), srB = Math.floor((this.y + this.h + 2) / TILE);
   185→   185→      for (let c = sc0; c <= sc1; c++) {
   186→   186→        if (tileAt(level, c, srA) === 'v' || tileAt(level, c, srB) === 'v') {
   187→   187→          this.vy = -1180; this.onGround = false; this.squash = -0.4; this.springT = 0.6;
   188→   188→          Audio8.sfx.spring();
   189→   189→          game.spawnDust(this.x + this.w / 2, this.y + this.h, 6);
   190→   190→          break;
   191→   191→        }
   192→   192→      }
   193→   193→    }
   194→   194→
   195→   195→    // падение в яму
   196→   196→    if (this.y > ROWS * TILE + 60) game.playerFell();
   197→   197→
   198→   198→    if (this.invuln > 0) this.invuln -= dt;
   199→   199→    if (Math.abs(this.vx) > 20 && this.onGround) this.walkT += dt * Math.abs(this.vx) / 26;
   200→   200→  }
   201→   201→
   202→   202→  hurt(dir, game, fromSpike = false) {
   203→   203→    if (this.invuln > 0) return;
   204→   204→    this.hp--;
   205→   205→    this.invuln = 1.3;
   206→   206→    this.vy = fromSpike ? -430 : -330;
   207→   207→    this.vx = dir * 260;
   208→   208→    Audio8.sfx.hurt();
   209→   209→    game.onPlayerHurt();
   210→   210→  }

   212→   212→  respawn() {
   213→   213→    this.x = this.lastSafe.x; this.y = this.lastSafe.y;
   214→   214→    this.vx = 0; this.vy = 0;
   215→   215→    this.invuln = 1.3;
   216→   216→  }
