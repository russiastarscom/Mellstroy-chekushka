   150→function defToState(def) {
   151→  const w = def.width;
   152→  const cells = Array.from({ length: ROWS }, () => Array(w).fill('.'));
   153→  (def.ground || []).forEach(([c0, c1]) => {
   154→    for (let r = 11; r < ROWS; r++) for (let c = c0; c <= c1; c++) if (c >= 0 && c < w) cells[r][c] = '#';
   155→  });
   156→  (def.bricks || []).forEach(([c0, r0, c1, r1]) => {
   157→    for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++)
   158→      if (r >= 0 && r < ROWS && c >= 0 && c < w) cells[r][c] = 'B';
   159→  });
   160→  (def.plats || []).forEach(([c, r, len]) => {
   161→    for (let i = 0; i < len; i++) { const cc = c + i; if (r >= 0 && r < ROWS && cc >= 0 && cc < w) cells[r][cc] = '='; }
   162→  });
   163→  (def.spikes || []).forEach(([c0, c1, r]) => {
   164→    const row = (r === undefined || r === null) ? 10 : r;
   165→    for (let c = c0; c <= c1; c++) if (row >= 0 && row < ROWS && c >= 0 && c < w) cells[row][c] = '^';
   166→  });
   167→  const ents = [];
   168→  const add = (t, c, r) => { if (c >= 0 && c < w && r >= 0 && r < ROWS) ents.push({ t, c, r }); };
   169→  (def.bottles?.runs || []).forEach(([c0, c1, r]) => { for (let c = c0; c <= c1; c++) add('bottle', c, r); });
   170→  (def.bottles?.singles || []).forEach(([c, r]) => add('bottle', c, r));
   171→  (def.hearts || []).forEach(([c, r]) => add('heart', c, r));
   172→  (def.plushes || []).forEach(([c, r]) => add('plush', c, r));
   173→  (def.enemies || []).forEach(([t, c, r]) => add(t, c, (r === undefined || r === null) ? 10 : r));
   174→  if (def.boss) add('G', def.boss.col, 9);
   175→  return {
   176→    cells, ents, undo: [], dirty: false,
   177→    spawn: def.spawn?.c ?? 2, factory: def.factory?.c ?? (w - 6),
   178→    name: def.name || ORIGINALS_NAME(def),
   179→  };
   180→}

   183→function stateToDef(idx) {
   184→  const orig = ORIGINALS[idx];
   185→  const st = E[idx];
   186→  const { cells, ents } = st;
   187→  const w = orig.width;
   188→  const ground = [], bricks = [], plats = [], spikes = [];
   189→  const singles = [], hearts = [], plushes = [], enemies = [];
   190→  let bossCol = null;
   191→  for (let r = 11; r < ROWS; r++) {
   192→    let c = 0;
   193→    while (c < w) {
   194→      if (cells[r][c] === '#') {
   195→        const c0 = c; while (c < w && cells[r][c] === '#') c++;
   196→        if (!ground.some((g) => g[0] === c0 && g[1] === c - 1)) ground.push([c0, c - 1]);
   197→      }
   198→      else c++;
   199→    }
   200→  }
   201→  for (let r = 0; r < ROWS; r++) {
   202→    let c = 0;
   203→    while (c < w) {
   204→      const ch = cells[r][c];
   205→      if (ch === 'B' || ch === '=' || ch === '^') {
   206→        const c0 = c; while (c < w && cells[r][c] === ch) c++;
   207→        if (ch === 'B') bricks.push([c0, r, c - 1, r]);
   208→        else if (ch === '=') plats.push([c0, r, c - c0]);
   209→        else spikes.push([c0, c - 1, r]);
   210→      } else c++;
   211→    }
   212→  }
   213→  ents.forEach((e) => {
   214→    if (e.t === 'bottle') singles.push([e.c, e.r]);
   215→    else if (e.t === 'heart') hearts.push([e.c, e.r]);
   216→    else if (e.t === 'plush') plushes.push([e.c, e.r]);
   217→    else if (e.t === 'e' || e.t === 't') enemies.push([e.t, e.c, e.r]);
   218→    else if (e.t === 'G') bossCol = e.c;
   219→  });
   220→  const def = {
   221→    id: orig.id, type: 'map', bg: orig.bg, width: w,
   222→    name: (st.name || orig.name || 'Карта').slice(0, 40),
   223→    ground, bricks, plats, spikes,
   224→    bottles: { runs: [], singles },
   225→    hearts, enemies,
   226→    spawn: { c: st.spawn }, factory: { c: st.factory },
   227→  };
   228→  if (orig.plushes || plushes.length) def.plushes = plushes;
   229→  if (bossCol !== null) def.boss = { col: bossCol };
   230→  if (orig.factoryLocked) def.factoryLocked = true;
   231→  if (orig.hints) def.hints = orig.hints;
   232→  if (orig.dialogue) def.dialogue = orig.dialogue;
   233→  return def;
   234→}

   236→function validateState(idx) {
   237→  const orig = ORIGINALS[idx];
   238→  const st = E[idx];
   239→  const w = orig.width;
   240→  if (st.spawn < 1 || st.spawn > w - 2) return 'Старт Андрея вне карты';

   243→  if (above !== '.') return `Старт (${st.spawn},10) занят тайлом «${above}» — убери его`;
   244→  for (const e of st.ents) {
   245→    if (e.c < 0 || e.c >= w) return 'Сущность вне карты';
   246→    if (e.r >= 11) return 'Сущность внутри земли — подними выше';
   247→  }
   248→  if (orig.factoryLocked && !st.ents.some((e) => e.t === 'G'))
   249→    return 'На карте 4 обязателен Вождь 👑 (без него завод не откроется)';
   250→  return null;
   251→}

   286→function draw() {
   287→  const st = E[cur];
   288→  const w = st.cells[0].length;
   289→  const cs = zoom.cs;
   290→  const TOP = 14;
   291→  const dpr = window.devicePixelRatio || 1;
   292→  cv.width = (w * cs) * dpr;
   293→  cv.height = (TOP + ROWS * cs) * dpr;
   294→  cv.style.width = (w * cs) + 'px';
   295→  cv.style.height = (TOP + ROWS * cs) + 'px';
   296→  const x = cv.getContext('2d');
   297→  x.setTransform(dpr, 0, 0, dpr, 0, 0);
   298→  x.fillStyle = '#1c1c30'; x.fillRect(0, 0, w * cs, TOP + ROWS * cs);
   299→  // номера колонок
   300→  x.fillStyle = '#55557a'; x.font = '9px Arial'; x.textAlign = 'center';
   301→  for (let c = 0; c < w; c += 5) x.fillText(c, c * cs + cs / 2, 10);
   302→  // тайлы
   303→  for (let r = 0; r < ROWS; r++) {
   304→    for (let c = 0; c < w; c++) {
   305→      const ch = st.cells[r][c];
   306→      const px = c * cs, py = TOP + r * cs;
   307→      if (ch === '#') { x.fillStyle = '#7a4a2b'; x.fillRect(px, py, cs, cs); x.fillStyle = '#4caf50'; x.fillRect(px, py, cs, Math.max(3, cs * .22)); }
   308→      else if (ch === 'B') { x.fillStyle = '#9c4a2d'; x.fillRect(px, py, cs, cs); x.strokeStyle = '#5d2c1a'; x.lineWidth = 1; x.strokeRect(px + .5, py + .5, cs - 1, cs - 1); x.beginPath(); x.moveTo(px, py + cs / 2); x.lineTo(px + cs, py + cs / 2); x.stroke(); }
   309→      else if (ch === '=') { x.fillStyle = '#b98a4a'; x.fillRect(px, py + cs * .2, cs, cs * .45); x.fillStyle = '#8a6234'; x.fillRect(px, py + cs * .5, cs, cs * .15); }
   310→      else if (ch === '^') { x.fillStyle = '#b9b9c9'; for (let k = 0; k < 2; k++) { x.beginPath(); x.moveTo(px + k * cs / 2, py + cs); x.lineTo(px + k * cs / 2 + cs / 4, py + cs * .25); x.lineTo(px + (k + 1) * cs / 2, py + cs); x.closePath(); x.fill(); } }
   311→      if (r === 11 && st.cells[r][c] === '.' && (st.cells[r - 1] && st.cells[r - 1][c]) !== '#') { x.fillStyle = '#22223a'; x.fillRect(px, py, cs, cs * 2 / 3); }
   312→    }
   313→  }
   314→  // сетка
   315→  x.strokeStyle = '#ffffff12'; x.lineWidth = 1;
   316→  for (let c = 0; c <= w; c++) { x.beginPath(); x.moveTo(c * cs + .5, TOP); x.lineTo(c * cs + .5, TOP + ROWS * cs); x.stroke(); }
   317→  for (let r = 0; r <= ROWS; r++) { x.beginPath(); x.moveTo(0, TOP + r * cs + .5); x.lineTo(w * cs, TOP + r * cs + .5); x.stroke(); }
   318→  // завод (рисуем до сущностей)
   319→  drawFactory(x, st.factory, cs, TOP, st);
   320→  // сущности
   321→  st.ents.forEach((e) => drawEnt(x, e, cs, TOP));
   322→  // старт
   323→  drawSpawn(x, st.spawn, cs, TOP);
   324→  // курсор
   325→  if (hover) { x.strokeStyle = '#ffd23f'; x.lineWidth = 2; x.strokeRect(hover.c * cs + 1, TOP + hover.r * cs + 1, cs - 2, cs - 2); }

   328→function imgFit(x, im, cx, bottomY, maxW, maxH) {
   329→  if (!im.complete || !im.naturalWidth) return false;
   330→  const k = Math.min(maxW / im.naturalWidth, maxH / im.naturalHeight);
   331→  const dw = im.naturalWidth * k, dh = im.naturalHeight * k;
   332→  x.drawImage(im, cx - dw / 2, bottomY - dh, dw, dh);
   333→  return true;
   334→}
   335→function drawEnt(x, e, cs, TOP) {

   361→function cellAt(ev) {
   362→  const rect = cv.getBoundingClientRect();
   363→  const cs = zoom.cs, TOP = 14;
   364→  const px = (ev.clientX - rect.left) * (cv.width / (window.devicePixelRatio || 1) / rect.width);

   371→function pushUndo() {
   372→  const st = E[cur];
   373→  st.undo.push(JSON.stringify({ cells: st.cells, ents: st.ents, spawn: st.spawn, factory: st.factory }));
   374→  if (st.undo.length > 40) st.undo.shift();

   376→function applyTool(c, r) {
   377→  const st = E[cur];
   378→  const w = st.cells[0].length;
   379→  const entAt = (cc, rr) => st.ents.find((e) => e.c === cc && e.r === rr);
   380→  if (tool === 'ground') {
   381→    if (r < 11) { toast('Земля — только нижние ряды 11–12. Выше ставь кирпич или платформу.'); return; }
   382→    if (st.cells[r][c] !== '#') { st.cells[11][c] = '#'; st.cells[12][c] = '#'; st.dirty = true; } // земля всегда 2 ряда, как в игре
   383→  } else if (tool === 'B' || tool === '=' || tool === '^') {
   384→    if (st.cells[r][c] !== tool) {
   385→      if (r >= 11 && tool !== 'B' && st.cells[r][c] === '#') { toast('Сначала сотри землю ластиком.'); return; }
   386→      st.cells[r][c] = tool; st.dirty = true;
   387→    }
   388→  } else if (tool === 'erase') {
   389→    const e = entAt(c, r);
   390→    if (e) { st.ents.splice(st.ents.indexOf(e), 1); st.dirty = true; }
   391→    if (st.cells[r][c] !== '.') {
   392→      if (r >= 11 && st.cells[r][c] === '#') { st.cells[11][c] = '.'; st.cells[12][c] = '.'; } // стираем яму насквозь
   393→      else st.cells[r][c] = '.';
   394→      st.dirty = true;
   395→    }
   396→  } else if (tool === 'spawn') {
   397→    if (st.cells[10][c] !== '.') { toast('Клетка старта занята тайлом — сотри его.'); return; }
   398→    if (st.spawn !== c) { st.spawn = c; st.dirty = true; }
   399→  } else if (tool === 'factory') {
   400→    if (st.factory !== c) { st.factory = c; st.dirty = true; }
   401→  } else {
   402→    // сущности
   403→    if (r >= 11) { toast('В землю сущности не ставятся.'); return; }
   404→    const e = entAt(c, r);
   405→    if (e && e.t === tool) return;
   406→    if (e) st.ents.splice(st.ents.indexOf(e), 1);
   407→    st.ents.push({ t: tool, c, r: tool === 'G' ? Math.min(r, 9) : r });
   408→    st.dirty = true;
   409→  }
   410→  draw(); buildTabs(); saveDrafts(); setStatus(pubAt);
   411→}

   511→async function boot() {
   512→  let ov = null;
   513→  try { const r = await fetch(API, { cache: 'no-store' }); const j = await r.json(); if (j && j.levels) ov = j; } catch (e) { /* нет API */ }
   514→  MAP_IDX.forEach(([i]) => { E[i] = defToState((ov && ov.levels && ov.levels[i]) || ORIGINALS[i]); });
   515→  // черновик браузера применяем, только если он не старше публикации с сервера
   516→  const dr = loadDrafts();
   517→  let appliedDraft = false;
   518→  let hadDirtyDraft = false;
   519→  if (dr) {
   520→    const pubTs = ov && ov.updatedAt ? (new Date(ov.updatedAt).getTime() || 0) : 0;
   521→    if (!pubTs || (dr.savedAt || 0) >= pubTs) {
   522→      appliedDraft = applyDrafts(dr);
   523→      hadDirtyDraft = appliedDraft && MAP_IDX.some(([i]) => E[i].dirty);
   524→    } else {
   525→      try { localStorage.removeItem(DRAFT_KEY); } catch (e) {}
   526→      setTimeout(() => toast('На сервере публикация новее — старый черновик отброшен', true), 600);
   527→    }
   528→  }
   529→  pubAt = (ov && ov.updatedAt) || null;
   530→  cur = MAP_IDX[0][0];
   531→  setStatus(pubAt);
   532→  $('lvlname').value = E[cur].name;
   533→  fitZoom(); buildTabs(); draw();
   534→  if (hadDirtyDraft) toast('Черновик восстановлен — есть НЕОПУБЛИКОВАННЫЕ правки', true);
   535→  else if (appliedDraft) toast('Черновик восстановлен из браузера', true);
   536→  if (ov) toast('Загружены ОПУБЛИКОВАННЫЕ карты (редактируешь их)', true);
   537→  window.ADMIN = { E, stateToDef, defToState, publish, buildPayload, paint: (c, r, t) => { tool = t; applyTool(c, r); }, get cur() { return cur; } };
   538→}
