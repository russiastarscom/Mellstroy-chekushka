     4→const Game = (() => {
     5→  const canvas = document.getElementById('game');
     6→  const ctx = canvas.getContext('2d');
     7→  ctx.imageSmoothingEnabled = false;
     8→
     9→  let VIEW_W = 960;
    10→  const VIEW_H = 540;
    11→  let ZOOM = 1.4;               // приближение камеры к персонажу
    12→  let camY = 0;                 // вертикальная камера (мир 540 по высоте)
    13→
    14→  // ---------- автрастягивание под весь экран ----------
    15→  // Высота мира всегда 540 (вертикальный геймплей не меняется),
    16→  // ширина обзора подгоняется под пропорции окна — чёрных полос нет.
    17→  function resizeCanvas() {
    18→    const vw = Math.max(1, window.innerWidth), vh = Math.max(1, window.innerHeight);
    19→    const aspect = Math.max(0.5, Math.min(3.2, vw / vh));
    20→    VIEW_W = Math.max(480, Math.min(1920, Math.round(VIEW_H * aspect)));
    21→    canvas.width = VIEW_W;
    22→    canvas.height = VIEW_H;
    23→    ctx.imageSmoothingEnabled = false;
    24→    // в портрете зум поменьше — нужно видеть дальше вперёд
    25→    ZOOM = VIEW_W >= VIEW_H ? 1.4 : 1.2;
    26→  }
    27→
    28→  // ---------- полный экран (Fullscreen API) ----------
    29→  // Тап по экрану только ОТКРЫВАЕТ полный экран — и всё.
    30→  // Обратно не выключаем: выйти можно штатными средствами браузера (Esc).
    31→  function isFullscreen() { return !!document.fullscreenElement; }
    32→  function enterFullscreen() {
    33→    try {
    34→      if (isFullscreen()) return; // уже во весь экран — ничего не делаем
    35→      const el = document.documentElement;
    36→      const fn = el.requestFullscreen || el.webkitRequestFullscreen;
    37→      if (fn) { const p = fn.call(el); if (p && p.catch) p.catch(() => {}); }
    38→    } catch (e) { /* игнор — некоторые окружения запрещают фуллскрин */ }
    39→  }
    40→
    41→  let state = 'boot';           // boot | menu | playing | paused | gameover | transition | ending
    42→  let frozen = false;           // true во время диалогов
    43→  let level = null, levelIndex = 0;
    44→  let player = null;
    45→  let enemies = [], tomahawks = [], bottles = [], hearts = [], particles = [];
    46→  let plushes = [], hatShots = [];
    47→  let customEnts = [];          // кастомные объекты из админ-панели (скриптуемые)
    48→  let factory = null;
    49→  let cam = 0, shake = 0;
    50→  let bottlesGot = 0, kills = 0;
    51→  let tGlobal = 0, lastT = 0;
    52→  let bossRef = null;
    53→  let hudCache = { hp: -1, bottles: -1, hats: -1 };
    54→
    55→  const input = { left: false, right: false, down: false, jumpHeld: false, jumpPressed: false, shootPressed: false };
    56→
    57→  // API для сущностей (передаётся вместо this)
    58→  const api = {
    59→    spawnDust, spawnStars, onPlayerHurt, playerFell, onBossDead,
    60→    get player() { return player; },
    61→    get tomahawks() { return tomahawks; },
    62→    get shake() { return shake; },
    63→    set shake(v) { shake = v; },
    64→    get kills() { return kills; },
    65→    set kills(v) { kills = v; },
    66→  };
    67→
    68→  // API для скриптов кастомных объектов (админ-панель)
    69→  const scriptApi = Object.create(api);
    70→  Object.assign(scriptApi, {
    71→    // гравитация + коллизии с тайлами (вызывай каждый кадр для «физических» объектов)
    72→    gravity(o, d) { o.vy = Math.min((o.vy || 0) + CONFIG.GRAVITY * d, 1100); moveEntity(o, level, d, false); },
    73→    // упёрся ли в стену на прошлом кадре
    74→    solid(o) { return !!o.hitWall; },
    75→    // живые враги на карте: у обычных нет hp — убиваются e.stomp(api), у босса есть hp — бьётся e.hitByHat(api)
    76→    enemies() { return enemies.filter(e => !e.dead); },
    77→    // чекушки на карте: b.taken — подобрана, b.rect() — прямоугольник {x,y,w,h}
    78→    bottles() { return bottles; },
    79→    // твёрдый ли тайл в пиксельной точке
    80→    solidAt(px, py) { return isSolid(level, Math.floor(px / TILE), Math.floor(py / TILE)); },
    81→    playerNear(o, r = 60) { return player ? Math.abs(player.x - o.x) < r && Math.abs(player.y - o.y) < r : false; },
    82→    hurtPlayer(dir = 1) { if (player) player.hurt(dir, api); },
    83→    healPlayer(n = 1) { if (player && player.hp < CONFIG.PLAYER_HP) { player.hp += n; Audio8.sfx.heart(); } },
    84→    ammo(n = 1) { if (player) player.ammo = Math.min(9, player.ammo + n); },
    85→    score(n = 1) { bottlesGot += n; },
    86→    sfx(name = 'coin') { const f = Audio8.sfx[name]; if (f) f(); },
    87→    particles(x, y, kind = 'dust', n = 5) { for (let i = 0; i < n; i++) particles.push(mkParticle(x, y, kind)); },
    88→    shake(v = 0.2) { shake = Math.max(shake, v); },
    89→  });
    90→  // ВАЖНО: px/py/time — живые геттеры. Object.assign вызвал бы их ОДИН раз при
    91→  // загрузке (player ещё null → навсегда 0), поэтому только defineProperties.
    92→  Object.defineProperties(scriptApi, {
    93→    px: { get() { return player ? player.x : 0; } },
    94→    py: { get() { return player ? player.y : 0; } },
    95→    time: { get() { return tGlobal; } },
    96→  });
    97→
    98→  // ---------- прогресс ----------
    99→  const SAVE_KEY = 'melstroy_chekushka_v1';
   100→  let progress = { unlocked: 1, done: {}, sound: true, musicVol: 1, sfxVol: 1 };
   101→  try {
   102→    const raw = localStorage.getItem(SAVE_KEY);
   103→    if (raw) progress = Object.assign(progress, JSON.parse(raw));
   104→  } catch (e) { /* ignore */ }
   105→  function saveProgress() {
   106→    try { localStorage.setItem(SAVE_KEY, JSON.stringify(progress)); } catch (e) { /* ignore */ }
   107→  }

   109→  // ---------- загрузка уровня ----------
   110→  function startLevel(i) {
   111→    levelIndex = i;
   112→    const def = LEVELS[i];
   113→    if (!def) return;
   114→
   115→    // прогресс: запоминаем, на каком уровне игрок находится
   116→    progress.cur = i;
   117→    saveProgress();
   118→
   119→    if (def.type === 'indev') { Audio8.stopMusic(); UI.showScreen('indev'); return; }
   120→
   121→    if (def.type === 'cutscene') {
   122→      state = 'transition';
   123→      Audio8.stopMusic();
   124→      UI.showScreen(null);
   125→      UI.setHud({ levelName: def.name, hp: CONFIG.PLAYER_HP, bottles: 0 });
   126→      UI.cutscene(def.cutscene, CUTSCENE_BG_BY_INDEX).then(async () => {
   127→        await finishLevel(true);
   128→      });
   129→      return;
   130→    }
   131→
   132→    // карта
   133→    level = buildLevel(def);
   134→    enemies = []; tomahawks = []; bottles = []; hearts = []; particles = [];
   135→    plushes = []; hatShots = [];
   136→    customEnts = [];
   137→    bossRef = null;
   138→    level.spawns.enemies.forEach(({ type, c, r, hp }) => {
   139→      if (type === 'e') enemies.push(new Walker(c, r));
   140→      else if (type === 't') enemies.push(new Thrower(c, r));
   141→      else if (type === 'f') enemies.push(new Flyer(c, r === undefined ? 6 : r));
   142→      else if (type === 'j') enemies.push(new Jumper(c, r));
   143→      else if (type === 'a') enemies.push(new Armored(c, r));
   144→      else if (type === 'boss') { const b = new Boss(c, { hp, big: hp >= 10 }); enemies.push(b); bossRef = b; }
   145→    });
   146→    level.spawns.bottles.forEach(({ c, r }) => bottles.push(new Bottle(c, r)));
   147→    level.spawns.hearts.forEach(({ c, r }) => hearts.push(new HeartPickup(c, r)));
   148→    (level.spawns.plushes || []).forEach(({ c, r }) => plushes.push(new PlushPickup(c, r)));
   149→    (level.spawns.custom || []).forEach(({ type, c, r }) => { if (CustomObjects.def(type)) customEnts.push(new CustomEnt(type, c, r)); });
   150→    factory = new FactoryExit(level.factoryC, !!def.factoryLocked);
   151→    player = new Player(level.spawnC);
   152→    player.hp = CONFIG.PLAYER_HP;
   153→    player.ammo = 0; player.shootCd = 0;   // боезапас плюшек
   154→    bottlesGot = 0;
   155→    kills = 0;
   156→    cam = 0; shake = 0;
   157→    camY = VIEW_H - VIEW_H / ZOOM; // сразу «прижата к полу»
   158→    hudCache = { hp: -1, bottles: -1, hats: -1 };
   159→    // сброс ввода — чтобы залипшая клавиша/кнопка не тянула Андрея после рестарта
   160→    input.left = false; input.right = false; input.down = false;
   161→    input.jumpHeld = false; input.jumpPressed = false; input.shootPressed = false;
   162→
   163→    state = 'playing';
   164→    frozen = true; // пока идёт вступительный диалог
   165→    UI.showScreen(null);
   166→    UI.setHud({ levelName: def.name, hp: player.hp, bottles: 0, hats: 0 });
   167→    Audio8.startMusic();
   168→
   169→    UI.dialogue(def.dialogue?.intro).then(() => { frozen = false; });
   170→  }
   171→
   172→  // ---------- завершение уровня ----------
   173→  async function finishLevel(isIntro = false) {
   174→    const def = LEVELS[levelIndex];
   175→    Audio8.stopMusic();
   176→    Audio8.sfx.win();
   177→    frozen = true;
   178→
   179→    if (!isIntro && def.dialogue?.outro) await UI.dialogue(def.dialogue.outro);
   180→
   181→    progress.done[levelIndex] = true;
   182→    progress.unlocked = Math.max(progress.unlocked, Math.min(levelIndex + 2, LEVELS.length - 1));
   183→    saveProgress();
   184→
   185→    // финальный экран — после последнего играбельного уровня (дальше только «В разработке» или конец)
   186→    const nextDef = LEVELS[levelIndex + 1];
   187→    if (!nextDef || nextDef.type === 'indev') {
   188→      state = 'ending';
   189→      UI.showScreen('ending');
   190→    } else if (levelIndex === 0) {
   191→      UI.showComplete('Вступление пройдено!<br>Дальше — первая карта Буримовки.', true);
   192→    } else {
   193→      const total = bottles.length;
   194→      UI.showComplete(`Чекушки собрано: <b style="color:#ffd23f">${bottlesGot}</b> / ${total}`, true);
   195→    }
   196→    state = 'transition';
   197→  }
   198→
   199→  // ---------- события ----------
   200→  function onPlayerHurt() {
   201→    if (player.hp <= 0) gameOver();
   202→  }
   203→  function playerFell() {
   204→    player.hp--;
   205→    shake = 0.35;
   206→    if (player.hp <= 0) { Audio8.sfx.hurt(); gameOver(); }
   207→    else { Audio8.sfx.hurt(); player.respawn(); }
   208→  }
   209→  function gameOver() {
   210→    state = 'gameover';
   211→    Audio8.stopMusic();
   212→    Audio8.sfx.lose();
   213→    UI.showScreen('gameover');
   214→  }
   215→  function onBossDead() {
   216→    // взрыв звёзд
   217→    for (let i = 0; i < 26; i++) particles.push(mkParticle(bossRef.x + bossRef.w / 2, bossRef.y + bossRef.h / 2, 'star'));
   218→    if (factory) factory.unlock();
   219→    frozen = true;
   220→    const outro = (LEVELS[levelIndex] && LEVELS[levelIndex].bossOutro) || DIALOGUES.outroAfterBoss;
   221→    UI.dialogue(outro).then(() => {
   222→      frozen = false;
   223→    });
   224→  }
   225→
   226→  // ---------- частицы ----------
   227→  function mkParticle(x, y, kind) {
   228→    const colors = { dust: '#c9c2b8', star: Math.random() < 0.5 ? '#ffd23f' : '#ff4757', sparkle: '#9fd8b4' };
   229→    return {
   230→      x, y,
   231→      vx: (Math.random() - 0.5) * 260,
   232→      vy: kind === 'dust' ? -Math.random() * 120 : (Math.random() - 0.7) * 300,
   233→      g: kind === 'dust' ? 120 : 300,
   234→      life: 0.5 + Math.random() * 0.4, maxLife: 0.9,
   235→      size: kind === 'star' ? 6 : 4,
   236→      color: colors[kind] || '#fff',
   237→    };
   238→  }
   239→  function spawnDust(x, y, n) { for (let i = 0; i < n; i++) particles.push(mkParticle(x, y, 'dust')); }
   240→  function spawnStars(x, y) { for (let i = 0; i < 10; i++) particles.push(mkParticle(x, y, 'star')); }
   241→
   242→  // ---------- обновление ----------
   243→  function update(dt) {
   244→    tGlobal += dt;
   245→    if (shake > 0) shake -= dt;
   246→
   247→    if (state !== 'playing' || frozen) { input.jumpPressed = false; return; }
   248→
   249→    player.update(dt, input, level, api);
   250→
   251→    // ---------- сыпучие платформы: триггер под ногами + обрушение ----------
   252→    if (player.onGround) {
   253→      const fr = Math.floor((player.y + player.h + 2) / TILE);
   254→      const c0 = Math.floor(player.x / TILE), c1 = Math.floor((player.x + player.w - 1) / TILE);
   255→      for (let c = c0; c <= c1; c++) {
   256→        if (tileAt(level, c, fr) === 'C' && !level.crumble.has(c + ',' + fr)) {
   257→          level.crumble.set(c + ',' + fr, { t: 0, broken: false });
   258→          Audio8.sfx.crumble();
   259→        }
   260→      }
   261→    }
   262→    level.crumble.forEach((st, k) => {
   263→      st.t += dt;
   264→      if (!st.broken && st.t > 0.45) {
   265→        st.broken = true; st.t = 0;
   266→        Audio8.sfx.crumble();
   267→        const [cc, rr] = k.split(',').map(Number);
   268→        spawnDust(cc * TILE + 20, rr * TILE + 20, 6);
   269→      } else if (st.broken && st.t > 2.6) {
   270→        level.crumble.delete(k);
   271→      }
   272→    });
   273→
   274→    // ---------- плюшки: бросок ----------
   275→    if (player.shootCd > 0) player.shootCd -= dt;
   276→    if (input.shootPressed) {
   277→      input.shootPressed = false;
   278→      if (player.ammo > 0 && player.shootCd <= 0) {
   279→        player.ammo--;
   280→        player.shootCd = 0.45;
   281→        const dir = player.dir || 1;
   282→        const hx = dir > 0 ? player.x + player.w - 4 : player.x - 26;
   283→        hatShots.push(new HatShot(hx, player.y + 2, dir));
   284→        Audio8.sfx.throw();
   285→      } else if (player.ammo <= 0) {
   286→        Audio8.sfx.locked(); // пусто — щёлкаем
   287→      }
   288→    }
   289→
   290→    enemies.forEach((e) => e.update(dt, level, api));
   291→    tomahawks.forEach((t) => t.update(dt, level, api));
   292→    tomahawks = tomahawks.filter((t) => !t.dead);
   293→
   294→    // столкновения игрок-враги
   295→    enemies.forEach((e) => {
   296→      if (e.dead) return;
   297→      if (!overlaps(player, e)) return;
   298→      const stomping = player.vy > 90 && (player.y + player.h) - e.y < 20;
   299→      if (stomping) {
   300→        if (e instanceof Boss) {
   301→          if (e.invuln > 0) { player.vy = -480; }
   302→          else { e.stomp(api); player.vy = -540; }
   303→        } else {
   304→          e.stomp(api);
   305→          player.vy = -490;
   306→        }
   307→      } else {
   308→        player.hurt(Math.sign(player.x - e.x) || 1, api);
   309→      }
   310→    });
   311→    enemies = enemies.filter((e) => !e.dead);
   312→
   313→    // предметы
   314→    bottles.forEach((b) => {
   315→      b.update(dt);
   316→      if (!b.taken && overlaps(player, b.rect())) {
   317→        b.taken = true;
   318→        bottlesGot++;
   319→        Audio8.sfx.coin();
   320→        for (let i = 0; i < 4; i++) particles.push(mkParticle(b.c * TILE + 20, b.r * TILE + 20, 'sparkle'));
   321→      }
   322→    });
   323→    bottles = bottles.filter((b) => !b.taken);
   324→    hearts.forEach((h) => {
   325→      h.update(dt);
   326→      if (!h.taken && overlaps(player, h.rect())) {
   327→        h.taken = true;
   328→        if (player.hp < CONFIG.PLAYER_HP) { player.hp++; Audio8.sfx.heart(); }
   329→        else { bottlesGot++; Audio8.sfx.coin(); }
   330→      }
   331→    });
   332→    hearts = hearts.filter((h) => !h.taken);
   333→
   334→    // плюшки — подбор боеприпаса
   335→    plushes.forEach((p) => {
   336→      p.update(dt);
   337→      if (!p.taken && overlaps(player, p.rect())) {
   338→        p.taken = true;
   339→        player.ammo = Math.min(9, player.ammo + 3);
   340→        Audio8.sfx.heart();
   341→        for (let i = 0; i < 6; i++) particles.push(mkParticle(p.c * TILE + 20, p.r * TILE + 20, 'sparkle'));
   342→      }
   343→    });
   344→    plushes = plushes.filter((p) => !p.taken);
   345→
   346→    // кастомные объекты (админ-панель): скрипт + взаимодействия с игроком
   347→    customEnts.forEach((o) => o.update(dt, level, scriptApi));
   348→    customEnts.forEach((o) => {
   349→      if (o.dead || o.taken) return;
   350→      if (!overlaps(player, o)) return;
   351→      const stomping = player.vy > 90 && (player.y + player.h) - o.y < 24;
   352→      if (o.collect > 0) {
   353→        o.taken = true;
   354→        bottlesGot += o.collect;
   355→        Audio8.sfx.coin();
   356→        scriptApi.particles(o.x + o.w / 2, o.y + o.h / 2, 'sparkle', 5);
   357→      } else if (stomping && o.stompable) {
   358→        o.hurt(api);
   359→        player.vy = -490;
   360→      } else if (o.dangerous) {
   361→        player.hurt(Math.sign(player.x - o.x) || 1, api);
   362→      }
   363→    });
   364→    customEnts = customEnts.filter((o) => !o.dead && !o.taken);
   365→
   366→    // летящие плюшки — столкновения
   367→    hatShots.forEach((s) => s.update(dt, level, api));
   368→    hatShots.forEach((s) => {
   369→      if (s.dead) return;
   370→      for (const e of enemies) {
   371→        if (e.dead) continue;
   372→        if (overlaps(s, e)) {
   373→          s.dead = true;
   374→          if (e instanceof Boss) e.hitByHat(api);
   375→          else if (e.armor) e.die(api);
   376→          else e.stomp(api);
   377→          break;
   378→        }
   379→      }
   380→      // плюшки бьют и скриптуемых объектов (вожди-кастомы и прочие stompable)
   381→      if (!s.dead) {
   382→        for (const o of customEnts) {
   383→          if (o.dead || o.taken || !o.stompable) continue;
   384→          if (overlaps(s, o)) { s.dead = true; o.hurt(api); break; }
   385→        }
   386→      }
   387→    });
   388→    hatShots = hatShots.filter((s) => !s.dead);
   389→
   390→    // завод — выход (триггер = всё здание)
   391→    factory.update(dt, player);
   392→    if (!factory.locked && overlaps(player, factory.doorRect())) { finishLevel(); return; }
   393→
   394→    // частицы
   395→    particles.forEach((p) => { p.vy += p.g * dt; p.x += p.vx * dt; p.y += p.vy * dt; p.life -= dt; });
   396→    particles = particles.filter((p) => p.life > 0);
   397→
   398→    // камера: горизонт + мягкое вертикальное слежение, всё в «приближенных» координатах
   399→    const viewW = VIEW_W / ZOOM, viewH = VIEW_H / ZOOM;
   400→    const target = Math.max(0, Math.min(player.x + player.w / 2 - viewW * 0.44, level.w * TILE - viewW));
   401→    cam += (target - cam) * Math.min(1, dt * 9);
   402→    const ty = Math.max(0, Math.min(player.y + player.h / 2 - viewH * 0.55, VIEW_H - viewH));
   403→    camY += (ty - camY) * Math.min(1, dt * 7);
   404→
   405→    // HUD
   406→    if (hudCache.hp !== player.hp || hudCache.bottles !== bottlesGot || hudCache.hats !== player.ammo) {
   407→      hudCache = { hp: player.hp, bottles: bottlesGot, hats: player.ammo };
   408→      UI.setHud({ hp: player.hp, bottles: bottlesGot, hats: player.ammo });
   409→    }
   410→
   411→    input.jumpPressed = false;
   412→  }

   415→  const BG_COLORS = {
   416→    bg_fields: ['#87ceeb', '#5aa84f'],
   417→    bg_city: ['#9fb4c7', '#6b7a8c'],

   421→    bg_desert: ['#ffd9a0', '#e8a75a'],

   458→  function drawTiles() {
   459→    const viewW = VIEW_W / ZOOM;
   460→    const c0 = Math.max(0, Math.floor(cam / TILE) - 1);
   461→    const c1 = Math.min(level.w - 1, Math.ceil((cam + viewW) / TILE) + 1);
   462→    const T = theme(LEVELS[levelIndex].bg);
   463→    for (let r = 0; r < ROWS; r++) {
   464→      for (let c = c0; c <= c1; c++) {
   465→        const t = level.grid[r][c];
   466→        if (t === '.') continue;
   467→        let x = c * TILE - cam, y = r * TILE;
   468→        if (t === '#') {
   469→          const grass = !TILE_SOLID.has(tileAt(level, c, r - 1));
   470→          ctx.fillStyle = T.ground;
   471→          ctx.fillRect(x, y, TILE, TILE);
   472→          ctx.fillStyle = T.dark;
   473→          ctx.fillRect(x + 4, y + 14, 8, 6);
   474→          ctx.fillRect(x + 24, y + 26, 10, 6);
   475→          if (grass) {
   476→            ctx.fillStyle = T.top;
   477→            ctx.fillRect(x, y, TILE, 12);
   478→            ctx.fillStyle = T.top2;
   479→            ctx.fillRect(x, y, TILE, 5);
   480→          }
   481→        } else if (t === 'B') {
   482→          ctx.fillStyle = T.brick;
   483→          ctx.fillRect(x, y, TILE, TILE);
   484→          ctx.fillStyle = T.brickD;
   485→          ctx.fillRect(x, y + 19, TILE, 2);
   486→          ctx.fillRect(x, y, 2, TILE);
   487→          ctx.fillRect(x + 19, y + 19, 2, 21);
   488→          ctx.fillRect(x + 39, y, 1, TILE);
   489→        } else if (t === '=') {
   490→          ctx.fillStyle = T.plat;
   491→          ctx.fillRect(x, y, TILE, 16);
   492→          ctx.fillStyle = T.platT;
   493→          ctx.fillRect(x, y, TILE, 6);
   494→          ctx.fillStyle = T.platD;
   495→          ctx.fillRect(x, y + 14, TILE, 2);
   496→        } else if (t === 'C') {
   497→          // сыпучая платформа: трясётся после триггера, исчезает после обрушения

   580→  function render() {
   581→    ctx.clearRect(0, 0, VIEW_W, VIEW_H);

   584→    const def = LEVELS[levelIndex];

   588→    let sx = 0, sy = 0;
   589→    if (shake > 0) { sx = (Math.random() - 0.5) * 12 * shake; sy = (Math.random() - 0.5) * 12 * shake; }
   590→    ctx.translate(sx, sy);
   591→    // приближение: мир рисуется крупнее, камера следит за Андреем
   592→    ctx.scale(ZOOM, ZOOM);
   593→    ctx.translate(0, -camY);
   594→
   595→    drawTiles();
   596→    if (def.type === 'map') drawHints();
   597→
   598→    bottles.forEach((b) => b.draw(ctx, cam));
   599→    hearts.forEach((h) => h.draw(ctx, cam));
   600→    plushes.forEach((p) => p.draw(ctx, cam));
   601→    factory.draw(ctx, cam);
   602→    enemies.forEach((e) => e.draw(ctx, cam, tGlobal));
   603→    tomahawks.forEach((t) => t.draw(ctx, cam));
   604→    hatShots.forEach((s) => s.draw(ctx, cam));
   605→    customEnts.forEach((o) => o.draw(ctx, cam));
   606→    if (player) player.draw(ctx, cam);
   607→
   608→    // частицы
   609→    particles.forEach((p) => {
   610→      ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
   611→      ctx.fillStyle = p.color;
   612→      ctx.fillRect(p.x - cam - p.size / 2, p.y - p.size / 2, p.size, p.size);
   613→      ctx.globalAlpha = 1;
   614→    });
   615→
   616→    ctx.restore();
   617→  }
