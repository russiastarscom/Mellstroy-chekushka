     4→const TILE_SOLID = new Set(['#', 'B']);
     5→
     6→function isSolid(level, c, r) {
     7→  if (c < 0 || c >= level.w) return true;   // стены по краям уровня
     8→  if (r < 0 || r >= ROWS) return false;     // сверху/внизу — открыто
     9→  return TILE_SOLID.has(level.grid[r][c]);
    10→}
    11→function isPlat(level, c, r) {
    12→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return false;
    13→  const t = level.grid[r][c];
    14→  if (t === '=') return true;
    15→  // сыпучая платформа: твёрдая, пока не обрушилась
    16→  if (t === 'C') {
    17→    const st = level.crumble && level.crumble.get(c + ',' + r);
    18→    return !(st && st.broken);
    19→  }
    20→  return false;
    21→}
    22→function isSpike(level, c, r) {
    23→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return false;
    24→  return level.grid[r][c] === '^';
    25→}
    26→function tileAt(level, c, r) {
    27→  if (c < 0 || c >= level.w || r < 0 || r >= ROWS) return '.';
    28→  return level.grid[r][c];
    29→}
    30→function isLiquid(level, c, r) { return tileAt(level, c, r) === 'L'; }
    31→
    32→// Движение с разрешением коллизий по тайлам
    33→function moveEntity(ent, level, dt, ignorePlats = false) {
    34→  // --- X ---
    35→  ent.x += ent.vx * dt;
    36→  ent.hitWall = false;
    37→  {
    38→    const r0 = Math.floor(ent.y / TILE), r1 = Math.floor((ent.y + ent.h - 1) / TILE);
    39→    const c1 = Math.floor((ent.x + ent.w - 1) / TILE), c0 = Math.floor(ent.x / TILE);
    40→    if (ent.vx > 0) {
    41→      for (let r = r0; r <= r1; r++) if (isSolid(level, c1, r)) { ent.x = c1 * TILE - ent.w; ent.vx = 0; ent.hitWall = true; break; }
    42→    } else if (ent.vx < 0) {
    43→      for (let r = r0; r <= r1; r++) if (isSolid(level, c0, r)) { ent.x = (c0 + 1) * TILE; ent.vx = 0; ent.hitWall = true; break; }
    44→    }
    45→  }
    46→  // --- Y ---
    47→  const prevBottom = ent.y + ent.h;
    48→  ent.y += ent.vy * dt;
    49→  ent.onGround = false;
    50→  {
    51→    const c0 = Math.floor(ent.x / TILE), c1 = Math.floor((ent.x + ent.w - 1) / TILE);
    52→    const r0 = Math.floor(ent.y / TILE), r1 = Math.floor((ent.y + ent.h - 1) / TILE);
    53→    if (ent.vy > 0) {
    54→      for (let c = c0; c <= c1; c++) {
    55→        if (isSolid(level, c, r1)) { ent.y = r1 * TILE - ent.h; ent.vy = 0; ent.onGround = true; break; }
    56→        if (!ignorePlats && isPlat(level, c, r1) && prevBottom <= r1 * TILE + 8) {
    57→          ent.y = r1 * TILE - ent.h; ent.vy = 0; ent.onGround = true; break;
    58→        }
    59→      }
    60→    } else if (ent.vy < 0) {
    61→      for (let c = c0; c <= c1; c++) {
    62→        if (isSolid(level, c, r0)) { ent.y = (r0 + 1) * TILE; ent.vy = 0; break; }
    63→      }
    64→    }
    65→  }
    66→}
    67→
    68→function overlaps(a, b) {
    69→  return a.x < b.x + b.w && a.x + a.w > b.x && a.y < b.y + b.h && a.y + a.h > b.y;
    70→}

    75→class Player {
    76→  constructor(colC) {
    77→    this.w = 30; this.h = 50;
    78→    this.x = colC * TILE + 5;
    79→    this.y = 9 * TILE;
    80→    this.vx = 0; this.vy = 0;
    81→    this.dir = 1;
    82→    this.onGround = false;
    83→    this.hp = CONFIG.PLAYER_HP;
    84→    this.invuln = 0;
    85→    this.jumpBuffer = 0;
    86→    this.coyote = 0;
    87→    this.dropTimer = 0;
    88→    this.lastSafe = { x: this.x, y: this.y };
    89→    this.safeTimer = 0;
    90→    this.walkT = 0;
    91→    this.squash = 0;
    92→  }
    93→
    94→  update(dt, input, level, game) {
    95→    const SPEED = CONFIG.MOVE_SPEED, ACC = 2200, FRICTION = 2600;
    96→
    97→    // горизонталь
    98→    if (input.left) { this.vx -= ACC * dt; this.dir = -1; }
    99→    if (input.right) { this.vx += ACC * dt; this.dir = 1; }
   100→    if (!input.left && !input.right) {
   101→      const s = Math.sign(this.vx);
   102→      this.vx -= s * FRICTION * dt;
   103→      if (Math.sign(this.vx) !== s) this.vx = 0;
   104→    }
   105→    this.vx = Math.max(-SPEED, Math.min(SPEED, this.vx));
   106→
   107→    // прыжок: буфер + койот
   108→    this.jumpBuffer = input.jumpPressed ? 0.12 : Math.max(0, this.jumpBuffer - dt);
   109→    this.coyote = this.onGround ? 0.1 : Math.max(0, this.coyote - dt);
   110→    this.dropTimer = Math.max(0, this.dropTimer - dt);
   111→
   112→    if (this.jumpBuffer > 0 && input.down && this.onGround) {
   113→      // спрыгнуть с платформы
   114→      this.dropTimer = 0.22;
   115→      this.jumpBuffer = 0;
   116→      this.vy = 60;
   117→    } else if (this.jumpBuffer > 0 && (this.coyote > 0)) {
   118→      this.vy = CONFIG.JUMP_VEL;
   119→      this.jumpBuffer = 0;
   120→      this.coyote = 0;
   121→      this.squash = -0.25;
   122→      Audio8.sfx.jump();
   123→      game.spawnDust(this.x + this.w / 2, this.y + this.h, 4);
   124→    }
   125→    // короткий прыжок при отпускании (батут не режем — у него свой импульс-таймер)
   126→    this.springT = Math.max(0, (this.springT || 0) - dt);
   127→    if (!input.jumpHeld && this.springT <= 0 && this.vy < -260) this.vy = -260;
   128→
   129→    this.vy = Math.min(this.vy + CONFIG.GRAVITY * dt, 1100);
   130→    moveEntity(this, level, dt, this.dropTimer > 0);
   131→
   132→    // приземление — пыль + сквош
   133→    if (this.onGround && this.prevVy > 380) {
   134→      this.squash = 0.3;
   135→      game.spawnDust(this.x + this.w / 2, this.y + this.h, 3);
   136→    }
   137→    this.prevVy = this.vy;
   138→    this.squash *= Math.pow(0.001, dt);
   139→
   140→    // безопасная точка (для респауна после ямы)
   141→    this.safeTimer -= dt;
   142→    if (this.onGround && this.safeTimer <= 0 && this.vy === 0) {
   143→      const fc = Math.floor((this.x + this.w / 2) / TILE);
   144→      const fr = Math.floor((this.y + this.h + 6) / TILE);
   145→      if (isSolid(level, fc, fr)) { this.lastSafe = { x: this.x, y: this.y }; this.safeTimer = 0.4; }
   146→    }
   147→
   148→    // шипы
   149→    if (this.invuln <= 0) {
   150→      const c0 = Math.floor((this.x + 6) / TILE), c1 = Math.floor((this.x + this.w - 6) / TILE);
   151→      const r = Math.floor((this.y + this.h - 4) / TILE);
   152→      for (let c = c0; c <= c1; c++) {
   153→        if (isSpike(level, c, r)) { this.hurt(c * TILE + 20 < this.x + this.w / 2 ? 1 : -1, game, true); break; }
   154→      }
   155→    }
   156→
   157→    // жидкость (полынья/лава/кислота): касание = урон + респавн, как падение в яму
   158→    if (this.invuln <= 0) {
   159→      const c0 = Math.floor((this.x + 4) / TILE), c1 = Math.floor((this.x + this.w - 4) / TILE);
   160→      const r0 = Math.floor((this.y + 8) / TILE), r1 = Math.floor((this.y + this.h - 1) / TILE);
   161→      outer: for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++) {
   162→        if (isLiquid(level, c, r)) { game.playerFell(); return; }
   163→      }
   164→    }
   165→
   166→    // пилы: круглая пила в тайле — задевание с любой стороны
   167→    if (this.invuln <= 0) {
   168→      const c0 = Math.floor((this.x - 10) / TILE), c1 = Math.floor((this.x + this.w + 10) / TILE);
   169→      const r0 = Math.floor((this.y - 10) / TILE), r1 = Math.floor((this.y + this.h + 10) / TILE);
   170→      for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++) {
   171→        if (tileAt(level, c, r) !== 'S') continue;
   172→        const sx = c * TILE + 20, sy = r * TILE + 26, R = 24;
   173→        const nx = Math.max(this.x, Math.min(sx, this.x + this.w));
   174→        const ny = Math.max(this.y, Math.min(sy, this.y + this.h));
   175→        if ((nx - sx) * (nx - sx) + (ny - sy) * (ny - sy) < R * R) {
   176→          this.hurt(sx < this.x + this.w / 2 ? 1 : -1, game, true); break;
   177→        }
   178→      }
   179→    }
   180→
   181→    // батут: приземлился/наступил — подброс (тайл батута в клетке ног или под ногами)
   182→    if (this.onGround && this.vy === 0) {
   183→      const sc0 = Math.floor((this.x + 4) / TILE), sc1 = Math.floor((this.x + this.w - 4) / TILE);
   184→      const srA = Math.floor((this.y + this.h - 2) / TILE), srB = Math.floor((this.y + this.h + 2) / TILE);
   185→      for (let c = sc0; c <= sc1; c++) {
   186→        if (tileAt(level, c, srA) === 'v' || tileAt(level, c, srB) === 'v') {
   187→          this.vy = -1180; this.onGround = false; this.squash = -0.4; this.springT = 0.6;
   188→          Audio8.sfx.spring();
   189→          game.spawnDust(this.x + this.w / 2, this.y + this.h, 6);
   190→          break;
   191→        }
   192→      }
   193→    }
   194→
   195→    // падение в яму
   196→    if (this.y > ROWS * TILE + 60) game.playerFell();
   197→
   198→    if (this.invuln > 0) this.invuln -= dt;
   199→    if (Math.abs(this.vx) > 20 && this.onGround) this.walkT += dt * Math.abs(this.vx) / 26;
   200→  }
   201→
   202→  hurt(dir, game, fromSpike = false) {
   203→    if (this.invuln > 0) return;
   204→    this.hp--;
   205→    this.invuln = 1.3;
   206→    this.vy = fromSpike ? -430 : -330;
   207→    this.vx = dir * 260;
   208→    Audio8.sfx.hurt();
   209→    game.onPlayerHurt();
   210→  }
   211→
   212→  respawn() {
   213→    this.x = this.lastSafe.x; this.y = this.lastSafe.y;
   214→    this.vx = 0; this.vy = 0;
   215→    this.invuln = 1.3;
   216→  }
   217→
   218→  draw(ctx, camX) {
   219→    if (this.invuln > 0 && Math.floor(this.invuln * 12) % 2 === 0) return; // мигание
   220→    const bob = (Math.abs(this.vx) > 20 && this.onGround) ? Math.abs(Math.sin(this.walkT * 6)) * 3 : 0;
   221→    const sq = this.squash;
   222→    const w = this.w + 10 - Math.abs(sq) * 12;
   223→    const h = this.h - sq * 10;
   224→    const x = this.x - (w - this.w) / 2 - camX;
   225→    const y = this.y + (this.h - h) - bob;
   226→    // тень
   227→    ctx.fillStyle = 'rgba(0,0,0,.25)';
   228→    ctx.beginPath();
   229→    ctx.ellipse(this.x + this.w / 2 - camX, this.y + this.h + 3, 16, 4, 0, 0, 7);
   230→    ctx.fill();
   231→    Sprites.draw(ctx, 'andrey', x, y, w, h, this.dir < 0);
   232→  }
   233→}

   238→class Walker {
   239→  constructor(colC, r = 10) {
   240→    this.w = 34; this.h = 46;
   241→    this.x = colC * TILE + 3;
   242→    this.y = r * TILE - this.h + TILE;
   243→    this.vx = -60; this.vy = 0;
   244→    this.dir = -1;
   245→    this.dead = false;
   246→    this.walkT = Math.random() * 3;
   247→  }
   248→  update(dt, level, game) {
   249→    this.vy = Math.min(this.vy + CONFIG.GRAVITY * dt, 900);
   250→    moveEntity(this, level, dt);
   251→    if (this.hitWall) this.flip();
   252→    // не падать с края и не ходить в шипы
   253→    if (this.onGround) {
   254→      const frontX = this.vx > 0 ? this.x + this.w + 3 : this.x - 3;
   255→      const fc = Math.floor(frontX / TILE);
   256→      const fr = Math.floor((this.y + this.h + 8) / TILE);
   257→      if (!isSolid(level, fc, fr) && !isPlat(level, fc, fr)) this.flip();
   258→      const br = Math.floor((this.y + this.h - 6) / TILE);
   259→      if (isSpike(level, fc, br)) this.flip();
   260→    }
   261→    this.walkT += dt * 5;
   262→  }
   263→  flip() { this.vx = -Math.sign(this.vx || this.dir) * 60; this.dir = Math.sign(this.vx); }
   264→  stomp(game) {
   265→    this.dead = true;
   266→    Audio8.sfx.stomp();
   267→    game.spawnStars(this.x + this.w / 2, this.y + 8);
   268→    game.kills++;
   269→  }
   270→  draw(ctx, camX) {
   271→    ctx.fillStyle = 'rgba(0,0,0,.25)';
   272→    ctx.beginPath();
   273→    ctx.ellipse(this.x + this.w / 2 - camX, this.y + this.h + 3, 14, 4, 0, 0, 7);
   274→    ctx.fill();
   275→    const bob = Math.abs(Math.sin(this.walkT)) * 2.5;
   276→    Sprites.draw(ctx, 'burmaldenets', this.x - camX, this.y - bob, this.w + 10, this.h + 12 - 0, this.vx > 0);
   277→  }
   278→}

   304→class Tomahawk {
   305→  constructor(x, y, dir) {
   306→    this.w = 18; this.h = 18;
   307→    this.x = x - 9; this.y = y;
   308→    this.vx = dir * 250; this.vy = -300;
   309→    this.rot = 0; this.dead = false;
   310→    this.harmless = 0.12; // чтобы не бить прямо из руки в упор
   311→  }
   312→  update(dt, level, game) {
   313→    this.vy += 1000 * dt;
   314→    this.x += this.vx * dt;
   315→    this.y += this.vy * dt;
   316→    this.rot += dt * 14;
   317→    this.harmless -= dt;
   318→    const c = Math.floor((this.x + 9) / TILE), r = Math.floor((this.y + 9) / TILE);
   319→    if (isSolid(level, c, r)) { this.dead = true; game.spawnDust(this.x + 9, this.y + 9, 4); }
   320→    if (this.y > ROWS * TILE + 40) this.dead = true;
   321→    if (this.harmless <= 0 && this.overlapPlayer(game.player)) {
   322→      this.dead = true;
   323→      game.player.hurt(Math.sign(game.player.x - this.x) || 1, game);
   324→    }
   325→  }
   326→  overlapPlayer(p) {
   327→    return overlaps(this, p);
   328→  }
   329→  draw(ctx, camX) {
   330→    ctx.save();
   331→    ctx.translate(this.x + 9 - camX, this.y + 9);
   332→    ctx.rotate(this.rot);
   333→    const spr = Sprites.get('tomahawk');
   334→    const src = spr.img || spr.fallback;
   335→    ctx.imageSmoothingEnabled = false;
   336→    ctx.drawImage(src, -10, -10, 20, 20);
   337→    ctx.restore();
   338→  }

   410→class Armored extends Walker {

   414→    this.vx = -26;

   418→  flip() { this.vx = -Math.sign(this.vx || this.dir) * 26; this.dir = Math.sign(this.vx); }
   419→  stomp(game) {
   420→    // броня: прыжок сверху ничего не даёт (игрок просто отскакивает)

   448→class Boss {
   449→  constructor(colC, opts = {}) {
   450→    this.w = opts.big ? 84 : 66;
   451→    this.h = opts.big ? 104 : 84;
   452→    this.x = colC * TILE + (TILE - this.w) / 2;
   453→    this.y = 8 * TILE + (opts.big ? -14 : 0);
   454→    this.vx = -70; this.vy = 0;
   455→    this.dir = -1;
   456→    this.hp = opts.hp || 6; this.maxHp = this.hp;
   457→    this.big = !!opts.big;
   458→    this.dead = false;
   459→    this.invuln = 0;
   460→    this.chargeT = 5;
   461→    this.state = 'walk';      // walk | telegraph | charge
   462→    this.telegraphT = 0;
   463→    this.walkT = 0;
   464→    this.hurtFlash = 0;
   465→  }
   466→  get speed() { return 70 + (this.maxHp - this.hp) * (this.big ? 11 : 22); }
   467→
   468→  update(dt, level, game) {
   469→    if (this.invuln > 0) this.invuln -= dt;
   470→    if (this.hurtFlash > 0) this.hurtFlash -= dt;
   471→
   472→    if (this.state === 'telegraph') {
   473→      this.telegraphT -= dt;
   474→      this.vx = 0;
   475→      if (this.telegraphT <= 0) { this.state = 'charge'; this.chargeDir = Math.sign(game.player.x - this.x) || 1; this.vx = this.chargeDir * this.speed * 2.4; }
   476→    } else if (this.state === 'charge') {
   477→      this.vx = this.chargeDir * this.speed * 2.4;
   478→      this.chargeT -= 0;
   479→      if (this.hitWall || (this.chargeDir > 0 && game.player.x < this.x && Math.random() < 0.01)) { this.state = 'walk'; this.chargeT = 4.5 + Math.random() * 2; }
   480→    } else {
   481→      this.chargeT -= dt;
   482→      if (this.chargeT <= 0) { this.state = 'telegraph'; this.telegraphT = 0.7; Audio8.sfx.throw(); }
   483→    }
   484→
   485→    this.vy = Math.min(this.vy + CONFIG.GRAVITY * dt, 900);
   486→    moveEntity(this, level, dt);
   487→    if (this.hitWall) {
   488→      if (this.state === 'charge') { this.state = 'walk'; this.chargeT = 4.5 + Math.random() * 2; game.shake = 0.25; }
   489→      this.vx = -Math.sign(this.vx || this.dir) * this.speed;
   490→    }
   491→    // не падать с края (в арене и так стены, но на всякий)
   492→    if (this.onGround && this.state === 'walk') {
   493→      const frontX = this.vx > 0 ? this.x + this.w + 3 : this.x - 3;
   494→      const fc = Math.floor(frontX / TILE), fr = Math.floor((this.y + this.h + 8) / TILE);
   495→      if (!isSolid(level, fc, fr) && !isPlat(level, fc, fr)) this.vx = -Math.sign(this.vx) * this.speed;
   496→    }
   497→    this.walkT += dt * 4;
   498→  }
   499→
   500→  stomp(game) {
   501→    if (this.invuln > 0) return;
   502→    this.hp--;
   503→    this.invuln = 1.1;
   504→    this.hurtFlash = 0.4;
   505→    game.shake = 0.3;
   506→    game.spawnStars(this.x + this.w / 2, this.y + 10);
   507→    Audio8.sfx.bossHit();
   508→    if (this.hp <= 0) {
   509→      this.dead = true;
   510→      Audio8.sfx.bossDie();
   511→      game.shake = 0.6;
   512→      game.onBossDead();
   513→    }
   514→  }
   515→
   516→  // Плюшкой киднули — урон без инвулна (каждая плюшка считается)
   517→  hitByHat(game) {
   518→    this.hp--;
   519→    this.hurtFlash = 0.4;
   520→    game.shake = 0.3;
   521→    game.spawnStars(this.x + this.w / 2, this.y + 10);
   522→    Audio8.sfx.bossHit();
   523→    if (this.hp <= 0) {
   524→      this.dead = true;
   525→      Audio8.sfx.bossDie();
   526→      game.shake = 0.6;
   527→      game.onBossDead();
   528→    }
   529→  }
   530→
   531→  draw(ctx, camX, t) {
   532→    // телеграф рывка — мигает
   533→    let flash = false;
   534→    if (this.state === 'telegraph') flash = Math.floor(t * 10) % 2 === 0;
   535→    if (this.hurtFlash > 0) flash = true;
   536→    ctx.fillStyle = 'rgba(0,0,0,.3)';
   537→    ctx.beginPath();
   538→    ctx.ellipse(this.x + this.w / 2 - camX, this.y + this.h + 4, 26, 6, 0, 0, 7);
   539→    ctx.fill();
   540→    if (flash) {
   541→      ctx.save();
   542→      ctx.globalAlpha = 0.45;
   543→      Sprites.draw(ctx, 'boss', this.x - camX - 4, this.y - 4, this.w + 14, this.h + 12, this.vx > 0);
   544→      ctx.restore();
   545→    }
   546→    const bob = Math.abs(Math.sin(this.walkT)) * 3;
   547→    Sprites.draw(ctx, 'boss', this.x - camX, this.y - bob, this.w + 10, this.h + 8, this.vx > 0);
   548→    // HP босса
   549→    if (!this.dead) {
   550→      const bw = 70, bx = this.x + this.w / 2 - bw / 2 - camX, by = this.y - 18;
   551→      ctx.fillStyle = '#00000090'; ctx.fillRect(bx - 1, by - 1, bw + 2, 8);
   552→      ctx.fillStyle = '#ff4757'; ctx.fillRect(bx, by, bw * (this.hp / this.maxHp), 6);
   553→    }
   554→  }
   555→}

   631→// КАСТОМНЫЕ ОБЪЕКТЫ (админ-панель): своя текстура + JS-скрипт
   632→// ============================================================
   633→// Скрипт объекта — тело функции, исполняется КАЖДЫЙ КАДР:
   634→//   function(obj, api, dt) { ... }
   635→// obj — сам объект (x, y, w, h, vx, vy, hp, t, data…),
   636→// api — доступ к игре, dt — секунды с прошлого кадра.
   637→const CustomObjects = {
   638→  defs: new Map(),
   639→  runs: new Map(),
   640→  broken: new Set(),   // объекты с ошибкой в скрипте — чтобы не спамить консоль
   641→  set(list) {
   642→    this.defs.clear(); this.runs.clear(); this.broken.clear();
   643→    (list || []).forEach((o) => {
   644→      if (!o || !o.id) return;
   645→      this.defs.set(o.id, o);
   646→      try {
   647→        this.runs.set(o.id, new Function('obj', 'api', 'dt', '"use strict";\n' + String(o.script || '')));
   648→      } catch (e) {
   649→        this.broken.add(o.id);
   650→        console.warn('Объект «' + (o.name || o.id) + '»: ошибка в скрипте — ' + e.message);
   651→      }
   652→      if (o.tex) Sprites.addSlot('obj-' + o.id, { w: o.w, h: o.h, url: o.tex, color: '#9a4de6', portrait: o.character ? [0, 0.66] : null });
   653→    });
   654→  },
   655→  def(id) { return this.defs.get(id) || null; },
   656→  get list() { return Array.from(this.defs.values()); },
   657→};

   659→class CustomEnt {
   660→  constructor(type, c, r) {
   661→    const d = CustomObjects.def(type) || {};
   662→    this.type = type;
   663→    this.w = Math.max(8, Math.min(400, d.w || 32));
   664→    this.h = Math.max(8, Math.min(400, d.h || 32));
   665→    this.x = c * TILE + (TILE - this.w) / 2;
   666→    this.y = (r + 1) * TILE - this.h;      // стоит на дне клетки
   667→    this.vx = 0; this.vy = 0;
   668→    this.dir = -1;
   669→    this.t = 0;                             // время жизни (сек)
   670→    this.hp = 1; this.maxHp = 1;
   671→    this.dead = false; this.taken = false;
   672→    // флаги поведения — скрипт задаёт/меняет:
   673→    this.collect = 0;        // >0: касание игроком = подбор, +collect к чекушкам
   674→    this.dangerous = false;  // касание игроком = урон
   675→    this.stompable = false;  // убивается прыжком сверху и плюшкой
   676→    this.onGround = false; this.hitWall = false;
   677→    this.flash = 0;
   678→    this.data = {};          // переменные скрипта
   679→  }
   680→  hurt(game) {
   681→    this.hp--;
   682→    this.flash = 0.25;
   683→    if (this.hp <= 0) {
   684→      this.dead = true;
   685→      game.kills++;
   686→      game.spawnStars(this.x + this.w / 2, this.y + this.h / 2);
   687→    }
   688→  }
   689→  update(dt, level, game) {
   690→    this.t += dt;
   691→    if (this.flash > 0) this.flash -= dt;
   692→    // упал за пределы карты (например, в яму) — убираем
   693→    if (this.y > ROWS * TILE + 80) this.dead = true;
   694→    const run = CustomObjects.runs.get(this.type);
   695→    if (!run || CustomObjects.broken.has(this.type)) return;
   696→    try { run(this, game, dt); }
   697→    catch (e) {
   698→      CustomObjects.broken.add(this.type);
   699→      console.warn('Объект «' + this.type + '»: ошибка выполнения скрипта — ' + e.message);
   700→    }
   701→    if (this.hp > this.maxHp) this.maxHp = this.hp;
   702→  }
   703→  draw(ctx, camX) {
   704→    if (!Sprites.get('obj-' + this.type)) return;
   705→    ctx.fillStyle = 'rgba(0,0,0,.2)';
   706→    ctx.beginPath();
   707→    ctx.ellipse(this.x + this.w / 2 - camX, this.y + this.h + 2, Math.max(6, this.w * 0.4), 3, 0, 0, 7);
   708→    ctx.fill();
   709→    if (this.flash > 0) ctx.globalAlpha = 0.55;
   710→    Sprites.draw(ctx, 'obj-' + this.type, this.x - camX, this.y, this.w, this.h, this.dir > 0);
   711→    ctx.globalAlpha = 1;
   712→    if (this.maxHp > 1 && !this.dead) {
   713→      const bw = Math.max(30, this.w * 0.8);
   714→      const bx = this.x + this.w / 2 - bw / 2 - camX, by = this.y - 10;
   715→      ctx.fillStyle = '#00000090'; ctx.fillRect(bx - 1, by - 1, bw + 2, 6);
   716→      ctx.fillStyle = '#ff4757'; ctx.fillRect(bx, by, bw * Math.max(0, this.hp / this.maxHp), 4);
   717→    }
   718→  }
