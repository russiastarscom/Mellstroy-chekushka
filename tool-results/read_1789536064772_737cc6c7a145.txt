    64→    64→function defToState(def) {
    65→    65→  const w = def.width;
    66→    66→  const cells = Array.from({ length: ROWS }, () => Array(w).fill('.'));
    67→    67→  (def.ground || []).forEach(([c0, c1]) => {
    68→    68→    for (let r = 11; r < ROWS; r++) for (let c = c0; c <= c1; c++) if (c >= 0 && c < w) cells[r][c] = '#';
    69→    69→  });
    70→    70→  (def.bricks || []).forEach(([c0, r0, c1, r1]) => {
    71→    71→    for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++)
    72→    72→      if (r >= 0 && r < ROWS && c >= 0 && c < w) cells[r][c] = 'B';
    73→    73→  });
    74→    74→  (def.plats || []).forEach(([c, r, len]) => {
    75→    75→    for (let i = 0; i < len; i++) { const cc = c + i; if (r >= 0 && r < ROWS && cc >= 0 && cc < w) cells[r][cc] = '='; }
    76→    76→  });
    77→    77→  (def.spikes || []).forEach(([c0, c1, r]) => {
    78→    78→    const row = (r === undefined || r === null) ? 10 : r;
    79→    79→    for (let c = c0; c <= c1; c++) if (row >= 0 && row < ROWS && c >= 0 && c < w) cells[row][c] = '^';
    80→    80→  });
    81→    81→  const ents = [];
    82→    82→  const add = (t, c, r) => { if (c >= 0 && c < w && r >= 0 && r < ROWS) ents.push({ t, c, r }); };
    83→    83→  (def.bottles?.runs || []).forEach(([c0, c1, r]) => { for (let c = c0; c <= c1; c++) add('bottle', c, r); });
    84→    84→  (def.bottles?.singles || []).forEach(([c, r]) => add('bottle', c, r));
    85→    85→  (def.hearts || []).forEach(([c, r]) => add('heart', c, r));
    86→    86→  (def.plushes || []).forEach(([c, r]) => add('plush', c, r));
    87→    87→  (def.enemies || []).forEach(([t, c, r]) => add(t, c, (r === undefined || r === null) ? 10 : r));
    88→    88→  if (def.boss) add('G', def.boss.col, 9);
    89→    89→  return {
    90→    90→    cells, ents, undo: [], dirty: false,
    91→    91→    spawn: def.spawn?.c ?? 2, factory: def.factory?.c ?? (w - 6),
    92→    92→    name: def.name || 'Карта', bg: def.bg || 'bg_fields',
    93→    93→    dialogs: {
    94→    94→      intro: (def.dialogue?.intro || []).map((l) => ({ who: l.who, text: l.text })),
    95→    95→      outro: (def.dialogue?.outro || []).map((l) => ({ who: l.who, text: l.text })),
    96→    96→      hints: (def.hints || []).map((h) => ({ c: h.c, text: h.text })),
    97→    97→    },
    98→    98→  };
    99→    99→}

   111→   111→function stateToDef(m) {
   112→   112→  const st = m.ed;
   113→   113→  if (st.isIndev) {
   114→   114→    return { id: m.id, type: 'indev', name: (st.name || 'В разработке').slice(0, 40) };
   115→   115→  }
   116→   116→  const { cells, ents } = st;
   117→   117→  const w = st.cells[0].length;
   118→   118→  const ground = [], bricks = [], plats = [], spikes = [];
   119→   119→  const singles = [], hearts = [], plushes = [], enemies = [];
   120→   120→  let bossCol = null;
   121→   121→  for (let r = 11; r < ROWS; r++) {
   122→   122→    let c = 0;
   123→   123→    while (c < w) {
   124→   124→      if (cells[r][c] === '#') {
   125→   125→        const c0 = c; while (c < w && cells[r][c] === '#') c++;
   126→   126→        if (!ground.some((g) => g[0] === c0 && g[1] === c - 1)) ground.push([c0, c - 1]);
   127→   127→      } else c++;
   128→   128→    }
   129→   129→  }
   130→   130→  for (let r = 0; r < ROWS; r++) {
   131→   131→    let c = 0;
   132→   132→    while (c < w) {
   133→   133→      const ch = cells[r][c];
   134→   134→      if (ch === 'B' || ch === '=' || ch === '^') {
   135→   135→        const c0 = c; while (c < w && cells[r][c] === ch) c++;
   136→   136→        if (ch === 'B') bricks.push([c0, r, c - 1, r]);
   137→   137→        else if (ch === '=') plats.push([c0, r, c - c0]);
   138→   138→        else spikes.push([c0, c - 1, r]);
   139→   139→      } else c++;
   140→   140→    }
   141→   141→  }
   142→   142→  ents.forEach((e) => {
   143→   143→    if (e.t === 'bottle') singles.push([e.c, e.r]);
   144→   144→    else if (e.t === 'heart') hearts.push([e.c, e.r]);
   145→   145→    else if (e.t === 'plush') plushes.push([e.c, e.r]);
   146→   146→    else if (e.t === 'e' || e.t === 't') enemies.push([e.t, e.c, e.r]);
   147→   147→    else if (e.t === 'G') bossCol = e.c;
   148→   148→  });
   149→   149→  const dlg = st.dialogs || { intro: [], outro: [], hints: [] };
   150→   150→  const def = {
   151→   151→    id: m.id, type: 'map', bg: st.bg || 'bg_fields', width: w,
   152→   152→    name: (st.name || 'Карта').slice(0, 40),
   153→   153→    ground, bricks, plats, spikes,
   154→   154→    bottles: { runs: [], singles },
   155→   155→    hearts, enemies,
   156→   156→    spawn: { c: st.spawn }, factory: { c: st.factory },
   157→   157→  };
   158→   158→  if (plushes.length) def.plushes = plushes;
   159→   159→  if (bossCol !== null) def.boss = { col: bossCol };
   160→   160→  if (m.def && m.def.factoryLocked) def.factoryLocked = true;
   161→   161→  if (dlg.intro.length || dlg.outro.length) {
   162→   162→    def.dialogue = { intro: dlg.intro, outro: dlg.outro };
   163→   163→  }
   164→   164→  if (dlg.hints.length) def.hints = dlg.hints.map((h) => ({ c: h.c, text: h.text }));
   165→   165→  return def;
   166→   166→}

   168→   168→function validateMap(m) {
   169→   169→  const st = m.ed;

   174→   174→  if (st.cells[10][st.spawn] !== '.') return `Старт (${st.spawn},10) занят тайлом — убери его`;

   216→   216→function applyDrafts(d) {
   217→   217→  let any = false;
   218→   218→  // черновик — источник истины по списку карт (включая добавленные/удалённые)
   219→   219→  const list = d.maps || [];
   220→   220→  const restored = [];
   221→   221→  list.forEach((s) => {
   222→   222→    const def = s.def ? JSON.parse(JSON.stringify(s.def)) : null;
   223→   223→    const ed = s.ed || {};
   224→   224→    // экран «В разработке» — отдельный вид уровня без террейна
   225→   225→    if (ed.isIndev || (def && def.type === 'indev')) {
   226→   226→      const nm = ed.name || (def && def.name) || 'В разработке';
   227→   227→      restored.push({ key: s.key || 'indev', id: s.id || 6, def: (def && def.type === 'indev') ? def : { id: s.id || 6, type: 'indev', name: nm }, ed: indevState(nm, ed.clean === false) });
   228→   228→      any = true;
   229→   229→      return;
   230→   230→    }
   231→   231→    const w = def ? def.width : (Array.isArray(ed.cells) && ed.cells[0] ? ed.cells[0].length : 0);
   232→   232→    let base = null;
   233→   233→    if (w >= 20 && Array.isArray(ed.cells) && ed.cells.length === ROWS && ed.cells[0] && ed.cells[0].length === w && Array.isArray(ed.ents)) {
   234→   234→      base = ed;
   235→   235→    } else if (def) {
   236→   236→      base = defToState(def); base.dirty = true; // битый ed — берём def как есть
   237→   237→    } else return; // ни валидного ed, ни def — карту пропускаем
   238→   238→    restored.push({
   239→   239→      key: s.key, id: s.id || 2, def,
   240→   240→      ed: {
   241→   241→        cells: base.cells, ents: base.ents, undo: [],
   242→   242→        spawn: (typeof base.spawn === 'number') ? base.spawn : 2,
   243→   243→        factory: (typeof base.factory === 'number') ? base.factory : (w - 6),
   244→   244→        name: base.name || 'Карта', bg: base.bg || 'bg_fields',
   245→   245→        dialogs: base.dialogs || { intro: [], outro: [], hints: [] },
   246→   246→        dirty: !ed.clean,
   247→   247→      },
   248→   248→    });
   249→   249→    any = true;
   250→   250→  });
   251→   251→  // старый черновик (сделан ДО появления управления «В разработке») — добавляем официальный экран
   252→   252→  if (d.listV !== 3 && !restored.some((m) => m.ed.isIndev)) {
   253→   253→    const iv = officialIndev();
   254→   254→    if (iv) { restored.push(iv); any = true; }
   255→   255→  }
   256→   256→  if (restored.length) CMS.maps = restored;
   257→   257→  if (d.intro && Array.isArray(d.intro.cutscene)) {
   258→   258→    CMS.intro = { cutscene: d.intro.cutscene, outroAfterBoss: d.intro.outroAfterBoss || [] };
   259→   259→    introDirty = !d.introClean; any = true;
   260→   260→  }
   261→   261→  if (Array.isArray(d.socials)) { CMS.socials = d.socials; socialsDirty = !d.socialsClean; any = true; }
   262→   262→  if (d.textures && typeof d.textures === 'object') { CMS.textures = d.textures; texturesDirty = !d.texturesClean; any = true; }
   263→   263→  if (d.musicActive !== undefined) { CMS.music.active = d.musicActive; musicDirty = !d.musicClean; }
   264→   264→  if (d.listClean !== undefined) mapsListDirty = !d.listClean;
   265→   265→  return any;
   266→   266→}

   489→   489→function applyTool(c, r) {
   490→   490→  const st = curEd();
   491→   491→  const w = st.cells[0].length;
   492→   492→  const entAt = (cc, rr) => st.ents.find((e) => e.c === cc && e.r === rr);
   493→   493→  if (tool === 'ground') {
   494→   494→    if (r < 11) { toast('Земля — только нижние ряды 11–12. Выше ставь кирпич или платформу.'); return; }
   495→   495→    if (st.cells[r][c] !== '#') { st.cells[11][c] = '#'; st.cells[12][c] = '#'; st.dirty = true; }
   496→   496→  } else if (tool === 'B' || tool === '=' || tool === '^') {
   497→   497→    if (st.cells[r][c] !== tool) {
   498→   498→      if (r >= 11 && tool !== 'B' && st.cells[r][c] === '#') { toast('Сначала сотри землю ластиком.'); return; }
   499→   499→      st.cells[r][c] = tool; st.dirty = true;
   500→   500→    }
   501→   501→  } else if (tool === 'erase') {
   502→   502→    const e = entAt(c, r);
   503→   503→    if (e) { st.ents.splice(st.ents.indexOf(e), 1); st.dirty = true; }
   504→   504→    if (st.cells[r][c] !== '.') {
   505→   505→      if (r >= 11 && st.cells[r][c] === '#') { st.cells[11][c] = '.'; st.cells[12][c] = '.'; }
   506→   506→      else st.cells[r][c] = '.';
   507→   507→      st.dirty = true;
   508→   508→    }
   509→   509→  } else if (tool === 'spawn') {
   510→   510→    if (st.cells[10][c] !== '.') { toast('Клетка старта занята тайлом — сотри его.'); return; }
   511→   511→    if (st.spawn !== c) { st.spawn = c; st.dirty = true; }
   512→   512→  } else if (tool === 'factory') {
   513→   513→    if (st.factory !== c) { st.factory = c; st.dirty = true; }
   514→   514→  } else {
   515→   515→    if (r >= 11) { toast('В землю сущности не ставятся.'); return; }
   516→   516→    const e = entAt(c, r);
   517→   517→    if (e && e.t === tool) return;
   518→   518→    if (e) st.ents.splice(st.ents.indexOf(e), 1);
   519→   519→    st.ents.push({ t: tool, c, r: tool === 'G' ? Math.min(r, 9) : r });
   520→   520→    st.dirty = true;
   521→   521→  }
   522→   522→  draw(); buildMapTabs(); saveDrafts(); setStatus();
   523→   523→}
