    17→  function resizeCanvas() {
    18→    const vw = Math.max(1, window.innerWidth), vh = Math.max(1, window.innerHeight);

   109→  // ---------- загрузка уровня ----------
   110→  function startLevel(i) {
   111→    levelIndex = i;

   114→
   115→    // прогресс: запоминаем, на каком уровне игрок находится

   120→

   122→      state = 'transition';
   123→      Audio8.stopMusic();
   124→      UI.showScreen(null);
   125→      UI.setHud({ levelName: def.name, hp: CONFIG.PLAYER_HP, bottles: 0 });
   126→      UI.cutscene(def.cutscene, CUTSCENE_BG_BY_INDEX).then(async () => {

   148→    player = new Player(level.spawnC);

   152→    kills = 0;
   153→    cam = 0; shake = 0;
   154→    camY = VIEW_H - VIEW_H / ZOOM; // сразу «прижата к полу»
   155→    hudCache = { hp: -1, bottles: -1, hats: -1 };

   163→    UI.setHud({ levelName: def.name, hp: player.hp, bottles: 0, hats: 0 });
   164→    Audio8.startMusic();

   170→  async function finishLevel(isIntro = false) {
   171→    const def = LEVELS[levelIndex];
   172→    Audio8.stopMusic();
   173→    Audio8.sfx.win();
   174→    frozen = true;
   175→
   176→    if (!isIntro && def.dialogue?.outro) await UI.dialogue(def.dialogue.outro);
   177→
   178→    progress.done[levelIndex] = true;
   179→    progress.unlocked = Math.max(progress.unlocked, Math.min(levelIndex + 2, LEVELS.length - 1));
   180→    saveProgress();
   181→
   182→    // финальный экран — после последнего играбельного уровня (дальше только «В разработке» или конец)
   183→    const nextDef = LEVELS[levelIndex + 1];
   184→    if (!nextDef || nextDef.type === 'indev') {
   185→      state = 'ending';
   186→      UI.showScreen('ending');
   187→    } else if (levelIndex === 0) {
   188→      UI.showComplete('Вступление пройдено!<br>Дальше — первая карта Буримовки.', true);
   189→    } else {
   190→      const total = bottles.length;
   191→      UI.showComplete(`Чекушки собрано: <b style="color:#ffd23f">${bottlesGot}</b> / ${total}`, true);
   192→    }
   193→    state = 'transition';
   194→  }

   214→    for (let i = 0; i < 26; i++) particles.push(mkParticle(bossRef.x + bossRef.w / 2, bossRef.y + bossRef.h / 2, 'star'));
   215→    if (factory) factory.unlock();
   216→    frozen = true;
   217→    UI.dialogue(DIALOGUES.outroAfterBoss).then(() => {
   218→      frozen = false;
   219→    });
   220→  }

   222→  // ---------- частицы ----------
   223→  function mkParticle(x, y, kind) {
   224→    const colors = { dust: '#c9c2b8', star: Math.random() < 0.5 ? '#ffd23f' : '#ff4757', sparkle: '#9fd8b4' };
   225→    return {
   226→      x, y,
   227→      vx: (Math.random() - 0.5) * 260,
   228→      vy: kind === 'dust' ? -Math.random() * 120 : (Math.random() - 0.7) * 300,
   229→      g: kind === 'dust' ? 120 : 300,
   230→      life: 0.5 + Math.random() * 0.4, maxLife: 0.9,
   231→      size: kind === 'star' ? 6 : 4,
   232→      color: colors[kind] || '#fff',
   233→    };
   234→  }
   235→  function spawnDust(x, y, n) { for (let i = 0; i < n; i++) particles.push(mkParticle(x, y, 'dust')); }
   236→  function spawnStars(x, y) { for (let i = 0; i < 10; i++) particles.push(mkParticle(x, y, 'star')); }

   238→  // ---------- обновление ----------
   239→  function update(dt) {
   240→    tGlobal += dt;
   241→    if (shake > 0) shake -= dt;
   242→
   243→    if (state !== 'playing' || frozen) { input.jumpPressed = false; return; }
   244→
   245→    player.update(dt, input, level, api);
   246→
   247→    // ---------- плюшки: бросок ----------
   248→    if (player.shootCd > 0) player.shootCd -= dt;
   249→    if (input.shootPressed) {
   250→      input.shootPressed = false;
   251→      if (player.ammo > 0 && player.shootCd <= 0) {
   252→        player.ammo--;
   253→        player.shootCd = 0.45;
   254→        const dir = player.dir || 1;
   255→        const hx = dir > 0 ? player.x + player.w - 4 : player.x - 26;
   256→        hatShots.push(new HatShot(hx, player.y + 2, dir));
   257→        Audio8.sfx.throw();
   258→      } else if (player.ammo <= 0) {
   259→        Audio8.sfx.locked(); // пусто — щёлкаем
   260→      }
   261→    }
   262→
   263→    enemies.forEach((e) => e.update(dt, level, api));
   264→    tomahawks.forEach((t) => t.update(dt, level, api));
   265→    tomahawks = tomahawks.filter((t) => !t.dead);
   266→
   267→    // столкновения игрок-враги
   268→    enemies.forEach((e) => {
   269→      if (e.dead) return;
   270→      if (!overlaps(player, e)) return;
   271→      const stomping = player.vy > 90 && (player.y + player.h) - e.y < 20;
   272→      if (stomping) {
   273→        if (e instanceof Boss) {
   274→          if (e.invuln > 0) { player.vy = -480; }
   275→          else { e.stomp(api); player.vy = -540; }
   276→        } else {
   277→          e.stomp(api);
   278→          player.vy = -490;
   279→        }
   280→      } else {
   281→        player.hurt(Math.sign(player.x - e.x) || 1, api);
   282→      }
   283→    });
   284→    enemies = enemies.filter((e) => !e.dead);
   285→
   286→    // предметы
   287→    bottles.forEach((b) => {
   288→      b.update(dt);
   289→      if (!b.taken && overlaps(player, b.rect())) {
   290→        b.taken = true;
   291→        bottlesGot++;
   292→        Audio8.sfx.coin();
   293→        for (let i = 0; i < 4; i++) particles.push(mkParticle(b.c * TILE + 20, b.r * TILE + 20, 'sparkle'));
   294→      }
   295→    });
   296→    bottles = bottles.filter((b) => !b.taken);
   297→    hearts.forEach((h) => {
   298→      h.update(dt);
   299→      if (!h.taken && overlaps(player, h.rect())) {
   300→        h.taken = true;
   301→        if (player.hp < CONFIG.PLAYER_HP) { player.hp++; Audio8.sfx.heart(); }
   302→        else { bottlesGot++; Audio8.sfx.coin(); }
   303→      }
   304→    });
   305→    hearts = hearts.filter((h) => !h.taken);
   306→
   307→    // плюшки — подбор боеприпаса
   308→    plushes.forEach((p) => {
   309→      p.update(dt);
   310→      if (!p.taken && overlaps(player, p.rect())) {
   311→        p.taken = true;
   312→        player.ammo = Math.min(9, player.ammo + 3);
   313→        Audio8.sfx.heart();
   314→        for (let i = 0; i < 6; i++) particles.push(mkParticle(p.c * TILE + 20, p.r * TILE + 20, 'sparkle'));
   315→      }
   316→    });
   317→    plushes = plushes.filter((p) => !p.taken);
   318→
   319→    // кастомные объекты (админ-панель): скрипт + взаимодействия с игроком
   320→    customEnts.forEach((o) => o.update(dt, level, scriptApi));
   321→    customEnts.forEach((o) => {
   322→      if (o.dead || o.taken) return;
   323→      if (!overlaps(player, o)) return;
   324→      const stomping = player.vy > 90 && (player.y + player.h) - o.y < 24;
   325→      if (o.collect > 0) {
   326→        o.taken = true;
   327→        bottlesGot += o.collect;
   328→        Audio8.sfx.coin();
   329→        scriptApi.particles(o.x + o.w / 2, o.y + o.h / 2, 'sparkle', 5);
   330→      } else if (stomping && o.stompable) {
   331→        o.hurt(api);
   332→        player.vy = -490;
   333→      } else if (o.dangerous) {
   334→        player.hurt(Math.sign(player.x - o.x) || 1, api);
   335→      }
   336→    });
   337→    customEnts = customEnts.filter((o) => !o.dead && !o.taken);
   338→
   339→    // летящие плюшки — столкновения
   340→    hatShots.forEach((s) => s.update(dt, level, api));
   341→    hatShots.forEach((s) => {
   342→      if (s.dead) return;
   343→      for (const e of enemies) {
   344→        if (e.dead) continue;
   345→        if (overlaps(s, e)) {
   346→          s.dead = true;
   347→          if (e instanceof Boss) e.hitByHat(api);
   348→          else e.stomp(api);
   349→          break;
   350→        }
   351→      }
   352→      // плюшки бьют и скриптуемых объектов (вожди-кастомы и прочие stompable)
   353→      if (!s.dead) {
   354→        for (const o of customEnts) {
   355→          if (o.dead || o.taken || !o.stompable) continue;
   356→          if (overlaps(s, o)) { s.dead = true; o.hurt(api); break; }
   357→        }
   358→      }
   359→    });
   360→    hatShots = hatShots.filter((s) => !s.dead);
   361→
   362→    // завод — выход (триггер = всё здание)
   363→    factory.update(dt, player);
   364→    if (!factory.locked && overlaps(player, factory.doorRect())) { finishLevel(); return; }
   365→
   366→    // частицы
   367→    particles.forEach((p) => { p.vy += p.g * dt; p.x += p.vx * dt; p.y += p.vy * dt; p.life -= dt; });
   368→    particles = particles.filter((p) => p.life > 0);
   369→
   370→    // камера: горизонт + мягкое вертикальное слежение, всё в «приближенных» координатах
   371→    const viewW = VIEW_W / ZOOM, viewH = VIEW_H / ZOOM;
   372→    const target = Math.max(0, Math.min(player.x + player.w / 2 - viewW * 0.44, level.w * TILE - viewW));
   373→    cam += (target - cam) * Math.min(1, dt * 9);
   374→    const ty = Math.max(0, Math.min(player.y + player.h / 2 - viewH * 0.55, VIEW_H - viewH));
   375→    camY += (ty - camY) * Math.min(1, dt * 7);
   376→
   377→    // HUD
   378→    if (hudCache.hp !== player.hp || hudCache.bottles !== bottlesGot || hudCache.hats !== player.ammo) {
   379→      hudCache = { hp: player.hp, bottles: bottlesGot, hats: player.ammo };
   380→      UI.setHud({ hp: player.hp, bottles: bottlesGot, hats: player.ammo });
   381→    }
   382→
   383→    input.jumpPressed = false;
   384→  }

   394→  function drawBackground(bgKey) {
   395→    const spr = Sprites.get(bgKey);
   396→    if (spr && spr.img) {
   397→      const img = spr.img;
   398→      const scale = VIEW_H / img.height;
   399→      const w = img.width * scale;
   400→      let off = -((cam * 0.28) % w);
   401→      for (let x = off; x < VIEW_W; x += w) ctx.drawImage(img, x, 0, w, VIEW_H);
   402→    } else {
   403→      const [top, bottom] = BG_COLORS[bgKey] || BG_COLORS.bg_fields;
   404→      const grad = ctx.createLinearGradient(0, 0, 0, VIEW_H);
   405→      grad.addColorStop(0, top); grad.addColorStop(1, bottom);
   406→      ctx.fillStyle = grad;
   407→      ctx.fillRect(0, 0, VIEW_W, VIEW_H);
   408→    }
   409→  }

   411→  function drawTiles() {
   412→    const viewW = VIEW_W / ZOOM;
   413→    const c0 = Math.max(0, Math.floor(cam / TILE) - 1);
   414→    const c1 = Math.min(level.w - 1, Math.ceil((cam + viewW) / TILE) + 1);
   415→    for (let r = 0; r < ROWS; r++) {
   416→      for (let c = c0; c <= c1; c++) {
   417→        const t = level.grid[r][c];
   418→        if (t === '.') continue;
   419→        const x = c * TILE - cam, y = r * TILE;
   420→        if (t === '#') {
   421→          const grass = !TILE_SOLID.has(level.grid[r - 1]?.[c] || '.');
   422→          ctx.fillStyle = '#7a4a2b';
   423→          ctx.fillRect(x, y, TILE, TILE);
   424→          ctx.fillStyle = '#63391f';
   425→          ctx.fillRect(x + 4, y + 14, 8, 6);
   426→          ctx.fillRect(x + 24, y + 26, 10, 6);
   427→          if (grass) {
   428→            ctx.fillStyle = '#3fa34d';
   429→            ctx.fillRect(x, y, TILE, 12);

   477→  function render() {
   478→    ctx.clearRect(0, 0, VIEW_W, VIEW_H);
   479→    if (!level) return;
   480→
   481→    const def = LEVELS[levelIndex];
   482→    drawBackground(def.bg || 'bg_fields');
   483→
   484→    ctx.save();
   485→    let sx = 0, sy = 0;
   486→    if (shake > 0) { sx = (Math.random() - 0.5) * 12 * shake; sy = (Math.random() - 0.5) * 12 * shake; }
   487→    ctx.translate(sx, sy);
   488→    // приближение: мир рисуется крупнее, камера следит за Андреем
   489→    ctx.scale(ZOOM, ZOOM);
   490→    ctx.translate(0, -camY);
   491→
   492→    drawTiles();
   493→    if (def.type === 'map') drawHints();
   494→
   495→    bottles.forEach((b) => b.draw(ctx, cam));
   496→    hearts.forEach((h) => h.draw(ctx, cam));
   497→    plushes.forEach((p) => p.draw(ctx, cam));
   498→    factory.draw(ctx, cam);
   499→    enemies.forEach((e) => e.draw(ctx, cam, tGlobal));
   500→    tomahawks.forEach((t) => t.draw(ctx, cam));
   501→    hatShots.forEach((s) => s.draw(ctx, cam));
   502→    customEnts.forEach((o) => o.draw(ctx, cam));
   503→    if (player) player.draw(ctx, cam);
   504→
   505→    // частицы
   506→    particles.forEach((p) => {
   507→      ctx.globalAlpha = Math.max(0, p.life / p.maxLife);
   508→      ctx.fillStyle = p.color;
   509→      ctx.fillRect(p.x - cam - p.size / 2, p.y - p.size / 2, p.size, p.size);
   510→      ctx.globalAlpha = 1;
   511→    });
   512→
   513→    ctx.restore();
   514→  }

   672→  function applyCms(j) {

   692→        }
   693→        LEVELS.splice(0, LEVELS.length, intro, ...defs);
   694→      }
   695→    } else if (j.levels) {
