     1→// ============================================================
     2→// Уровни и сюжет
     3→// Легенда тайлов: '#' земля, 'B' кирпич, '=' платформа (можно спрыгнуть),
     4→// 'C' сыпучая платформа (обрушается), '^' шипы, 'L' жидкость (полынья/лава/кислота),
     5→// 'v' батут, 'S' пила, 'c' чекушка, 'H' сердечко, 'e' бурмалденец, 't' метатель,
     6→// 'f' летун, 'j' прыгун, 'a' щитоносец, 'G' вождь-босс, 'F' завод (выход), 'P' старт
     7→// ============================================================
     8→const ROWS = 13;          // высота карты в тайлах
     9→const TILE = 40;          // размер тайла
    10→
    11→function buildLevel(def) {
    12→  const w = def.width;
    13→  const grid = Array.from({ length: ROWS }, () => Array(w).fill('.'));
    14→  const spawns = { enemies: [], bottles: [], hearts: [], tomahawks: [], plushes: [], custom: [] };
    15→
    16→  // земля (2 ряда внизу)
    17→  (def.ground || []).forEach(([c0, c1]) => {
    18→    for (let r = 11; r < ROWS; r++) for (let c = c0; c <= c1; c++) grid[r][c] = '#';
    19→  });
    20→  // кирпичи: прямоугольники [c0, r0, c1, r1]
    21→  (def.bricks || []).forEach(([c0, r0, c1, r1]) => {
    22→    for (let r = r0; r <= r1; r++) for (let c = c0; c <= c1; c++) grid[r][c] = 'B';
    23→  });
    24→  // платформы [c, r, len]
    25→  (def.plats || []).forEach(([c, r, len]) => {
    26→    for (let i = 0; i < len; i++) if (r >= 0 && r < ROWS && c + i < w) grid[r][c + i] = '=';
    27→  });
    28→  // сыпучие платформы [c, r, len] — обрушаются, когда на них постояли
    29→  (def.crumble || []).forEach(([c, r, len]) => {
    30→    for (let i = 0; i < len; i++) if (r >= 0 && r < ROWS && c + i < w) grid[r][c + i] = 'C';
    31→  });
    32→  // шипы [c0, c1] — на земле (ряд 10), либо [c0, c1, ряд] — например на платформе
    33→  (def.spikes || []).forEach(([c0, c1, r]) => {
    34→    const row = (r === undefined || r === null) ? 10 : r;
    35→    if (row >= 0 && row < ROWS) for (let c = c0; c <= c1; c++) grid[row][c] = '^';
    36→  });
    37→  // жидкость [c0, c1] — дно ям (ряд 12), либо [c0, c1, ряд]
    38→  (def.liquids || []).forEach(([c0, c1, r]) => {
    39→    const row = (r === undefined || r === null) ? 12 : r;
    40→    if (row >= 0 && row < ROWS) for (let c = c0; c <= c1; c++) grid[row][c] = 'L';
    41→  });
    42→  // батуты [c] — на земле (ряд 10), либо [c, ряд]
    43→  (def.springs || []).forEach(([c, r]) => {
    44→    const row = (r === undefined || r === null) ? 10 : r;
    45→    if (row >= 0 && row < ROWS) grid[row][c] = 'v';
    46→  });
    47→  // пилы [c] — в земле (ряд 10), либо [c, ряд]
    48→  (def.saws || []).forEach(([c, r]) => {
    49→    const row = (r === undefined || r === null) ? 10 : r;
    50→    if (row >= 0 && row < ROWS) grid[row][c] = 'S';
    51→  });
    52→  // чекушка
    53→  (def.bottles?.runs || []).forEach(([c0, c1, r]) => {
    54→    for (let c = c0; c <= c1; c++) spawns.bottles.push({ c, r });
    55→  });
    56→  (def.bottles?.singles || []).forEach(([c, r]) => spawns.bottles.push({ c, r }));
    57→  // сердечки
    58→  (def.hearts || []).forEach(([c, r]) => spawns.hearts.push({ c, r }));
    59→  // плюшки (боеприпас для стрельбы по боссу и щитоносцам)
    60→  (def.plushes || []).forEach(([c, r]) => spawns.plushes.push({ c, r }));
    61→  // враги: [тип, колонка] — спавн на землю (ряд 10), либо [тип, колонка, ряд] — на платформу
    62→  (def.enemies || []).forEach(([type, c, r]) => spawns.enemies.push({ type, c, r: (r === undefined) ? 10 : r }));
    63→  if (def.boss) spawns.enemies.push({ type: 'boss', c: def.boss.col, r: 9, hp: def.boss.hp });
    64→  // кастомные объекты из админ-панели: { 'obj-id': [[c,r],...] }
    65→  if (def.custom && typeof def.custom === 'object') {
    66→    Object.entries(def.custom).forEach(([type, list]) => {
    67→      (list || []).forEach(([c, r]) => spawns.custom.push({ type, c, r }));
    68→    });
    69→  }
    70→
    71→  return {
    72→    def, w, grid, spawns,
    73→    crumble: new Map(),          // состояние сыпучих платформ 'c,r' -> {t, broken}
    74→    spawnC: def.spawn?.c ?? 2,
    75→    factoryC: def.factory?.c ?? w - 6,
    76→    hints: def.hints || [],
    77→  };
    78→}

   323→// ============================================================
   324→// ГЕНЕРАТОР УРОВНЕЙ (детерминированный, по seed главы)
   325→// ============================================================
   326→function mulberry32(seed) {
   327→  let a = seed >>> 0;
   328→  return function () {
   329→    a |= 0; a = (a + 0x6D2B79F5) | 0;
   330→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   331→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   332→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   333→  };
   334→}

   336→// Один сгенерированный уровень главы ch (0..19)
   337→function genLevelDef(ch, idx, id) {
   338→  const M = CHAPTER_META[ch];
   339→  const isBossMid = idx === 9, isBossFin = idx === 19;
   340→  const rng = mulberry32(ch * 7919 + idx * 104729 + 5);
   341→  const d = idx;
   342→  const R = (a, b) => a + Math.floor(rng() * (b - a + 1));
   343→
   344→  const def = {

   346→    ground: [], plats: [], bricks: [], spikes: [], springs: [], saws: [],
   347→    crumble: [], liquids: [],
   348→    bottles: { runs: [], singles: [] }, hearts: [], plushes: [], enemies: [],
   349→    spawn: { c: 2 }, factory: { c: 0 },
   350→  };
   351→
   352→  // ---------- БОССОВЫЕ АРЕНЫ (10-й и 20-й уровень главы) ----------
   353→  if (isBossMid || isBossFin) {
   354→    const w = 90;
   355→    def.width = w;
   356→    def.ground = [[0, w - 1]];

   359→    def.bottles.runs = [[8, 10, 10], [36, 38, 10], [50, 52, 10], [70, 72, 10], [24, 26, 7], [42, 44, 7]];
   360→    def.hearts = [[25, 7], [57, 7]];
   361→    def.plushes = [[20, 10], [40, 10], [55, 10], [62, 10]];
   362→    def.enemies = [['e', 14], ['e', 30], ['t', 48], ['e', 60]];

   364→    if (ch === 2 && isBossFin) def.enemies.push(['a', 50]);
   365→    if (ch === 3) { def.enemies.push(['f', 36, 5], ['a', 46]); }
   366→    def.boss = { col: 76, hp: isBossFin ? M.bossHp.fin : M.bossHp.mid };
   367→    def.factoryLocked = true;
   368→    def.factory = { c: w - 5 };
   369→    def.hints = [{ c: 30, text: 'Плюшки! Подбирай — кидай по вождю (X)' }];
   370→    def.dialogue = { intro: isBossFin ? M.finIntro : M.midBossIntro, outro: isBossFin ? M.finOutro : M.midBossOutro };
   371→    if (isBossFin) {
   372→      if (ch === 3) {
   373→        def.bg = 'bg_final';
   374→        def.name = M.names[19];
   375→        def.boss.hp = 12;
   376→        def.bossOutro = DIALOGUES.happyAfterBoss;
   377→        def.dialogue.outro = DIALOGUES.happyOutro;
   378→        def.hints = [{ c: 30, text: 'ШЕФ ВСЕХ БУРМАЛДЕНОВ! Плюшки (X) + прыжки сверху!' }];
   379→      }
   380→      def.plats.push([68, 6, 3]);
   381→    }
   382→    return def;
   383→  }
   384→
   385→  // ---------- ОБЫЧНЫЕ УРОВНИ ----------
   386→  const w = Math.min(170, 92 + d * 3 + R(0, 9));
   387→  def.width = w;
   388→  def.factory = { c: w - 4 };
   389→  const busy = new Set();          // занятые колонки ряда 10 (препятствия)
   390→  const mark = (c0, c1) => { for (let c = c0; c <= c1; c++) busy.add(c); };
   391→
   392→  // земля: стартовая площадка → сегменты с ямами → финальная площадка
   393→  const pitMax = M.pit + Math.min(2, Math.floor(d / 5));
   394→  const startPad = R(6, 8);
   395→  def.ground.push([0, startPad]);
   396→  let c = startPad + 1;
   397→  const pits = [], segs = [];
   398→  while (true) {
   399→    if (w - c < 22) break;                       // остаток целиком станет финальной площадкой
   400→    let pitW = (c > startPad + 5 && rng() < 0.6) ? R(2, pitMax) : 0;
   401→    if (pitW && c + pitW > w - 14) pitW = 0;
   402→    if (pitW) { pits.push([c, c + pitW - 1, pitW]); c += pitW; }
   403→    const seg = R(5, 12);
   404→    def.ground.push([c, c + seg - 1]);
   405→    segs.push([c, c + seg - 1]);
   406→    c += seg;
   407→    if (w - c < 8) {                             // добиваем хвост землёй до края
   408→      def.ground[def.ground.length - 1][1] = w - 1;
   409→      segs[segs.length - 1][1] = w - 1;
   410→      c = w;
   411→      break;
   412→    }
   413→  }
   414→  if (c < w) { def.ground.push([c, w - 1]); segs.push([c, w - 1]); }
   415→
   416→  // сегменты для препятствий/врагов: не трогаем зону завода и старт
   417→  const placeSegs = segs.filter(g => g[1] < w - 10 && g[0] > 8);
   418→
   419→  // жидкость в ямах
   420→  pits.forEach(([p0, p1]) => def.liquids.push([p0, p1]));
   421→
   422→  // сыпучие мосты через широкие ямы (со 2-й половины главы)
   423→  pits.forEach(([p0, p1, pw]) => {
   424→    if (pw >= 4 && d >= 4 && rng() < 0.6) def.crumble.push([p0, 10, pw]);
   425→  });
   426→
   427→  // батуты: у широких ям + случайные
   428→  const springCount = ch === 1 ? (idx >= 3 ? R(1, 2) : 0) : (ch === 2 ? (idx >= 2 ? R(1, 3) : 0) : R(2, 3));
   429→  let placed = 0;
   430→  pits.forEach(([p0, p1, pw]) => {
   431→    if (pw >= 4 && placed < springCount && !busy.has(p0 - 1)) {
   432→      def.springs.push([p0 - 1]); mark(p0 - 1, p0 - 1); placed++;
   433→    }
   434→  });
   435→  for (let i = placed; i < springCount && placeSegs.length; i++) {
   436→    const g = placeSegs[R(0, placeSegs.length - 1)];
   437→    const sc = R(g[0] + 1, Math.max(g[0] + 1, g[1] - 1));
   438→    if (!busy.has(sc)) { def.springs.push([sc]); mark(sc, sc); }
   439→  }
   440→
   441→  // пилы в земле
   442→  const sawCount = ch === 1 ? (d >= 6 ? R(0, 2) : 0) : ch === 2 ? (d >= 3 ? R(1, 3) : 0) : R(1, 4);
   443→  for (let i = 0; i < sawCount && placeSegs.length; i++) {
   444→    const g = placeSegs[R(0, placeSegs.length - 1)];
   445→    const sc = R(g[0] + 1, Math.max(g[0] + 1, g[1] - 1));
   446→    if (!busy.has(sc) && !busy.has(sc - 1) && !busy.has(sc + 1)) { def.saws.push([sc]); mark(sc - 1, sc + 1); }
   447→  }
   448→
   449→  // шипы
   450→  const spikeCount = Math.min(6, Math.floor(d / 3) + (rng() < 0.5 ? 1 : 0));
   451→  for (let i = 0; i < spikeCount && placeSegs.length; i++) {
   452→    const g = placeSegs[R(0, placeSegs.length - 1)];
   453→    const sc = R(g[0] + 1, Math.max(g[0] + 1, g[1] - 2));
   454→    if (!busy.has(sc) && !busy.has(sc + 1)) {
   455→      const wide = (d >= 6 && rng() < 0.4) ? 2 : 1;
   456→      def.spikes.push([sc, sc + wide - 1]); mark(sc, sc + wide);
   457→    }
   458→  }
   459→
   460→  // платформы (ряд 8 в основном, ряд 6 — только рядом с рядом 8)
   461→  const platCount = 3 + Math.floor(d / 3) + R(0, 2);
   462→  const row8Plats = [];
   463→  for (let i = 0; i < platCount && segs.length; i++) {
   464→    const g = segs[R(0, segs.length - 1)];
   465→    const pc = R(g[0], Math.max(g[0], g[1] - 4));
   466→    const len = R(3, 5);
   467→    let pr = 8;
   468→    if (rng() < 0.3 && row8Plats.some(([pc8]) => Math.abs(pc8 - pc) <= 5)) pr = 6;
   469→    def.plats.push([pc, pr, len]);
   470→    if (pr === 8) row8Plats.push([pc, len]);
   471→  }
   472→
   473→  // враги
   474→  const pool = ['e', 'e'];
   475→  if (d >= 2 || ch >= 2) pool.push('t');
   476→  if ((ch === 1 && d >= 5) || (ch === 2 && d >= 2) || ch === 3) pool.push('j', 'e');
   477→  if ((ch === 1 && d >= 8) || (ch === 2 && d >= 5) || (ch === 3 && d >= 2)) pool.push('f');
   478→  if ((ch === 2 && d >= 8) || (ch === 3 && d >= 6)) pool.push('a');
   479→  const eCount = Math.min(13, 3 + Math.floor(d * 0.55));
   480→  let hasArmored = false;
   481→  for (let i = 0; i < eCount; i++) {
   482→    const type = pool[R(0, pool.length - 1)];
   483→    if (type === 'f') {
   484→      def.enemies.push(['f', R(12, w - 14), R(4, 6)]);
   485→      continue;
   486→    }
   487→    if (type === 'a') hasArmored = true;
   488→    if (rng() < 0.25 && def.plats.length) {
   489→      const [pc, pr, len] = def.plats[R(0, def.plats.length - 1)];
   490→      def.enemies.push([type, pc + R(0, len - 1), pr - 1]);
   491→    } else if (placeSegs.length) {
   492→      const g = placeSegs[R(0, placeSegs.length - 1)];
   493→      const ec = R(Math.max(g[0], 10), Math.max(g[0] + 1, g[1] - 1));
   494→      if (!busy.has(ec)) { def.enemies.push([type, ec]); mark(ec, ec); }
   495→    }
   496→  }
   497→  if (hasArmored) { def.plushes.push([R(12, Math.max(13, w - 20)), 10], [R(12, Math.max(13, w - 20)), 10]); }
   498→
   499→  // чекушки: ряды на земле, на платформах, дуги над ямами
   500→  segs.forEach(([g0, g1], si) => {
   501→    if (si === 0 || rng() < 0.85) {
   502→      const b0 = Math.min(g0 + 1, g1 - 2);
   503→      def.bottles.runs.push([b0, b0 + R(0, 2), 10]);
   504→    }
   505→  });
   506→  def.plats.forEach(([pc, pr, len]) => {
   507→    if (rng() < 0.7) def.bottles.singles.push([pc + Math.floor(len / 2), pr - 1]);
   508→  });
   509→  pits.forEach(([p0, p1]) => {
   510→    def.bottles.singles.push([p0, 8]);
   511→    if (p1 > p0) def.bottles.singles.push([p1, 8]);
   512→  });
   513→
   514→  // сердечки
   515→  const hCount = R(1, 2);
   516→  for (let i = 0; i < hCount; i++) {
   517→    if (def.plats.length) {
   518→      const [pc, pr, len] = def.plats[R(0, def.plats.length - 1)];
   519→      def.hearts.push([pc + Math.floor(len / 2), pr - 1]);
   520→    } else {
   521→      def.hearts.push([R(20, Math.max(21, w - 20)), 10]);
   522→    }
   523→  }
   524→
   525→  // подсказки и диалоги
   526→  def.hints = (M.hints || []).filter(([hIdx]) => hIdx === idx).map(([, text]) => ({ c: 6, text }));
   527→  if (idx === 0) {
   528→    def.dialogue = { intro: M.intro, outro: M.outro0 || null };
   529→  } else {
   530→    const pair = M.pool[idx % M.pool.length];
   531→    def.dialogue = { intro: pair, outro: null };
   532→  }
   533→  return def;
   534→}

   539→  return list;
   540→}
