// Оверрайд текстов в levels.js
// ============================================================

(function applyGameDialogs() {
  try {
    const D = window.GAME_DIALOGS;

    if (
      !D ||
      typeof LEVELS === 'undefined' ||
      typeof DIALOGUES === 'undefined'
    ) {
      return;
    }

    const norm = (a) => (
      Array.isArray(a)
        ? a
            .filter((line) => line && line.text)
            .map((line) => ({
              who: line.who || 'narrator',
              text: line.text
            }))
        : null
    );

    // Мутируем массивы на месте:
    // на них уже могут ссылаться катсцены и боссы.
    const putInPlace = (dst, src) => {
      if (!Array.isArray(dst) || !Array.isArray(src)) return;

      const next = norm(src);
      if (!next) return;

      dst.splice(0, dst.length, ...next);
    };

    putInPlace(DIALOGUES.intro, D.common.intro);
    putInPlace(DIALOGUES.outroAfterBoss, D.common.outroAfterBoss);
    putInPlace(DIALOGUES.happyAfterBoss, D.common.happyAfterBoss);
    putInPlace(DIALOGUES.happyOutro, D.common.happyOutro);

    // Полная замена диалогов и подсказок по ID карты.
    Object.keys(D.byId || {}).forEach((key) => {
      const levelId = Number(key);

      const level = LEVELS.find(
        (item) =>
          item &&
          item.type === 'map' &&
          item.id === levelId
      );

      if (!level) return;

      const override = D.byId[key] || {};
      const dialogue = level.dialogue || (level.dialogue = {});

      const intro = norm(override.intro);

      if (intro) {
        dialogue.intro = intro;
      }

      if ('outro' in override) {
        const outro = norm(override.outro);

        // null сохраняется для специального финала босса.
        dialogue.outro = Array.isArray(override.outro)
          ? (outro || [])
          : override.outro;
      }

      if (Array.isArray(override.hints)) {
        level.hints = override.hints
          .filter(
            (hint) =>
              hint &&
              typeof hint.c === 'number' &&
              hint.text
          )
          .map((hint) => ({
            c: hint.c,
            text: hint.text
          }));
      }
    });
  } catch (error) {
    console.error('[dialogs.js]', error);
  }
})();
